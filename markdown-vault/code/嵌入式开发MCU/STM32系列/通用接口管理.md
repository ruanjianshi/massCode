## Fragment: common.h
```c_cpp
#ifndef COMMON_H
#define COMMON_H

/**
 * @file    common.h
 * @brief   与具体外设无关的公共返回码
 *
 * 所有驱动统一使用这里的错误码，避免每层自造一套 0/-1。
 * 约定：
 *   - 返回 0（COMMON_OK）表示成功；
 *   - 返回负数表示错误（枚举值就是负的）；
 *   - read/write 类接口成功时返回正数（实际字节数），失败返回负错误码。
 */

/* =====================================================================
 * ① 公共错误码
 * ===================================================================== */
typedef enum {
    COMMON_OK       = 0,     /* 成功 */
    COMMON_EINVAL   = -1,    /* 参数非法（空指针、越界、非法值） */
    COMMON_ESTATE   = -2,    /* 状态错误（未打开/重复打开/未初始化） */
    COMMON_ENOSPC   = -3,    /* 容量不足（表满/缓冲满） */
    COMMON_EEXIST   = -4,    /* 已存在（重名注册） */
    COMMON_ENOENT   = -5,    /* 不存在（未找到/已注销） */
    COMMON_ENOSYS   = -6,    /* 功能未实现（必选方法缺失） */
    COMMON_EBUSY    = -7,    /* 忙（资源被占用/后端 BUSY） */
    COMMON_ETIMEOUT = -8,    /* 超时 */
    COMMON_EIO      = -9     /* 通用 IO 错误 */
} common_result_t;

#endif /* COMMON_H */
```

## Fragment: device.h
```c_cpp
#ifndef DEVICE_H
#define DEVICE_H

/**
 * @file    device.h
 * @brief   简单的静态设备注册表
 *
 * 这里只负责“按名称找到设备”，不负责打开和读写。
 * 注册和注销应在初始化阶段完成，不支持 ISR 或并发调用。
 *
 * 设备对象的三要素：
 *   api    —— 类型化驱动的操作表（如 uart_ops_t）
 *   config —— 编译期固定的硬件配置（如 HAL 句柄地址）
 *   data   —— 每个实例独立的运行期状态
 */

#include "common.h"

/* =====================================================================
 * ① 可裁剪配置区
 * ===================================================================== */
#ifndef DEV_NAME_MAX
#define DEV_NAME_MAX 24   /* 设备名最大长度（含结尾 '\0'），过长直接拒绝 */
#endif

#ifndef DEV_MAX
#define DEV_MAX 8         /* 注册表容量（静态数组，无动态内存） */
#endif

/* =====================================================================
 * ② 设备类别
 * ===================================================================== */
typedef enum {
    DEV_CLASS_NONE = 0,   /* 未初始化 / 非法 */
    DEV_CLASS_UART,
    DEV_CLASS_I2C,
    DEV_CLASS_SPI,
    DEV_CLASS_SENSOR
} dev_class_t;

/* =====================================================================
 * ③ 设备对象（不透明句柄）
 *
 * 结构体定义留在 device.c 中，头文件只给前置声明：
 * 业务代码只能通过下面的公共接口访问，不能直接解引用内部字段。
 * ===================================================================== */
typedef struct dev_entry dev_entry_t;

/* =====================================================================
 * ④ 注册表接口
 * ===================================================================== */

/**
 * @brief  登记一个设备（初始化阶段调用）
 * @param  name     设备名（非空、不重复、长度 < DEV_NAME_MAX）
 * @param  class_id 设备类别（须在 DEV_CLASS_NONE 与 DEV_CLASS_SENSOR 之间）
 * @param  api      类型化驱动操作表（必选，不允许 NULL）
 * @param  config   编译期固定硬件配置（可为 NULL）
 * @param  data     实例运行期状态（可为 NULL）
 * @param  out      输出参数：成功时写入设备对象地址
 * @return COMMON_OK 成功；COMMON_EINVAL 参数非法；COMMON_EEXIST 重名；
 *         COMMON_ENOSPC 表满
 * @note   register 只登记不打开。打开/读写/关闭由具体驱动接口完成。
 */
int dev_register(const char *name, dev_class_t class_id,
                 const void *api, const void *config, void *data,
                 dev_entry_t **out);

/**
 * @brief  按名称查找设备
 * @param  name 设备名
 * @return 命中的设备对象；未找到或参数非法返回 NULL
 * @note   线性查找，容量固定 DEV_MAX，耗时 O(DEV_MAX)。
 */
dev_entry_t *dev_find(const char *name);

/**
 * @brief  注销一个设备（停止阶段调用）
 * @param  dev 要注销的设备对象
 * @return COMMON_OK 成功；COMMON_ENOENT 对象不属于注册表或已注销
 * @note   调用前驱动必须先释放自己的资源（关闭 DMA/中断/句柄），
 *         本函数只把表项清零，不做资源回收。
 */
int dev_unregister(dev_entry_t *dev);

/* =====================================================================
 * ⑤ 访问器
 * ===================================================================== */

/** @brief 读取设备类别（非法对象返回 DEV_CLASS_NONE） */
dev_class_t dev_get_class(const dev_entry_t *dev);

/** @brief 读取类型化操作表（如 uart_ops_t；非法对象返回 NULL） */
const void *dev_get_api(const dev_entry_t *dev);

/** @brief 读取编译期硬件配置（非法对象返回 NULL） */
const void *dev_get_config(const dev_entry_t *dev);

/** @brief 读取实例运行期状态（非法对象返回 NULL） */
void *dev_get_data(const dev_entry_t *dev);

/* =====================================================================
 * ⑥ 扩展预留
 * ===================================================================== */
/*
 * - dev_count / dev_at：设备枚举
 * - mutex：运行期并发注册
 * - refcount：防止使用中的设备被注销
 */

#endif /* DEVICE_H */

```

## Fragment: device.c
```c_cpp
/**
 * @file    device.c
 * @brief   静态设备注册表实现
 *
 * 设计要点：
 * - 固定容量静态数组，零动态内存，适合 MCU 启动阶段一次性注册；
 * - 表项用 used 标志区分“占用/空闲”，注销即整块清零；
 * - 所有对外操作先校验参数与对象归属（entry_valid），再执行；
 * - 返回统一错误码（common.h 的 COMMON_* 枚举）。
 */

#include <string.h>
#include "device.h"

/* =====================================================================
 * ① 私有结构（不透明，业务代码不可见）
 * ===================================================================== */
struct dev_entry {
    char name[DEV_NAME_MAX];  /* 设备名（'\0' 结尾） */
    dev_class_t class_id;     /* 设备类别 */
    const void *api;          /* 类型化驱动操作表 */
    const void *config;       /* 编译期硬件配置 */
    void *data;               /* 实例运行期状态 */
    unsigned char used;       /* 1=已占用 0=空闲 */
};

/* =====================================================================
 * ② 静态注册表
 * ===================================================================== */
static dev_entry_t devices[DEV_MAX];

/* =====================================================================
 * ③ 内部工具
 * ===================================================================== */

/* 设备名校验：非 NULL、非空、且长度不超过 DEV_NAME_MAX-1（留 '\0' 位置） */
static int name_valid(const char *name)
{
    size_t i;

    if (name == NULL || name[0] == '\0')
        return 0;
    for (i = 0; i < DEV_NAME_MAX; i++)
        if (name[i] == '\0')
            return 1;
    return 0;
}

/* 对象归属校验：指针必须指向注册表中的一个“已占用”表项 */
static dev_entry_t *entry_valid(const dev_entry_t *dev)
{
    int i;

    for (i = 0; i < DEV_MAX; i++)
        if (&devices[i] == dev && devices[i].used)
            return &devices[i];
    return NULL;
}

/* =====================================================================
 * ④ 注册表接口
 * ===================================================================== */

dev_entry_t *dev_find(const char *name)
{
    int i;

    if (!name_valid(name))
        return NULL;
    for (i = 0; i < DEV_MAX; i++)
        if (devices[i].used && strcmp(devices[i].name, name) == 0)
            return &devices[i];
    return NULL;
}

int dev_register(const char *name, dev_class_t class_id,
                 const void *api, const void *config, void *data,
                 dev_entry_t **out)
{
    int i;

    /* 先清空输出参数，失败时不遗留脏指针 */
    if (out != NULL)
        *out = NULL;
    /* 参数校验：名称合法、类别在有效范围、操作表与输出指针必填 */
    if (!name_valid(name) || class_id <= DEV_CLASS_NONE ||
        class_id > DEV_CLASS_SENSOR ||
        api == NULL || out == NULL)
        return COMMON_EINVAL;
    /* 重名拒绝：注册表按名唯一 */
    if (dev_find(name) != NULL)
        return COMMON_EEXIST;

    /* 找第一个空闲槽并写入 */
    for (i = 0; i < DEV_MAX; i++) {
        if (!devices[i].used) {
            memset(&devices[i], 0, sizeof(devices[i]));
            memcpy(devices[i].name, name, strlen(name) + 1U);
            devices[i].class_id = class_id;
            devices[i].api = api;
            devices[i].config = config;
            devices[i].data = data;
            devices[i].used = 1U;
            *out = &devices[i];
            return COMMON_OK;
        }
    }
    return COMMON_ENOSPC;   /* 表满 */
}

int dev_unregister(dev_entry_t *dev)
{
    dev_entry_t *entry = entry_valid(dev);

    if (entry == NULL)
        return COMMON_ENOENT;
    memset(entry, 0, sizeof(*entry));   /* 整块清零 = 注销 */
    return COMMON_OK;
}

/* =====================================================================
 * ⑤ 访问器
 *
 * 所有访问器都先做归属校验：传入的对象不在注册表中时返回“空值”，
 * 避免悬垂指针被误用。
 * ===================================================================== */

dev_class_t dev_get_class(const dev_entry_t *dev)
{
    dev_entry_t *entry = entry_valid(dev);

    return entry == NULL ? DEV_CLASS_NONE : entry->class_id;
}

const void *dev_get_api(const dev_entry_t *dev)
{
    dev_entry_t *entry = entry_valid(dev);

    return entry == NULL ? NULL : entry->api;
}

const void *dev_get_config(const dev_entry_t *dev)
{
    dev_entry_t *entry = entry_valid(dev);

    return entry == NULL ? NULL : entry->config;
}

void *dev_get_data(const dev_entry_t *dev)
{
    dev_entry_t *entry = entry_valid(dev);

    return entry == NULL ? NULL : entry->data;
}

## Fragment: README.md
```markdown
# 通用接口管理（简化版）

目标只有一个：让业务代码通过名称找到设备，同时不依赖驱动内部结构。

## 基本关系

```text
设备名 → dev_entry → 类型化驱动 API + 硬件配置 + 实例数据
```

注册表只做三件事：

1. `dev_register`：初始化阶段登记设备；
2. `dev_find`：按名称查找；
3. `dev_unregister`：停止设备后注销。

## 使用示例

```c
dev_entry_t *entry = NULL;

if (dev_register("uart-debug", DEV_CLASS_UART,
                 &uart_default_ops, &huart1, uart_context,
                 &entry) != COMMON_OK) {
    /* 记录初始化错误 */
}
```

设备的打开、读写和关闭由 UART/I2C/SPI 等类型化接口完成，通用注册表不做 `void *` 函数调用。

## 代码分割与注释约定

每个文件按逻辑区块划分，区段标题用 `/* ==== ① xxx ==== */` 分隔：

- `common.h`：① 公共错误码；
- `device.h`：① 配置区 → ② 设备类别 → ③ 不透明对象 → ④ 注册表接口 → ⑤ 访问器 → ⑥ 扩展预留；
- `device.c`：① 私有结构 → ② 静态表 → ③ 内部工具 → ④ 接口实现 → ⑤ 访问器。

所有公共函数带 `@brief/@param/@return/@note` 注释；私有函数只写关键逻辑
（“为什么这么做”），不堆砌“做什么”的废话。

## 当前约束

- 固定容量，无动态内存；
- 名称过长直接返回错误，不静默截断；
- 只在初始化或停止阶段注册、注销；
- 不支持 ISR 和并发修改。

设备枚举、引用计数、锁和自动初始化顺序属于后续扩展，当前不进入基本实现。
