## Fragment: device.h
```c_cpp
#ifndef DEVICE_H
#define DEVICE_H

/**
 * @file    device.h
 * @brief   通用接口管理 —— 设备注册表（与具体驱动解耦）
 *
 * 驱动的"登记处"：驱动就绪后把设备注册进来，业务代码只认名字
 * （dev_find），不关心后端与硬件句柄。驱动通过 ops / hw / priv
 * 三个字段保存自己的信息并自行解释；本注册表本身不做任何 I/O。
 *
 * 串口驱动配套的便捷接口见串口通信驱动片段的 uart_dev.c。
 */

#include <stdint.h>

/* ==================== ① 配置区（可裁剪） ==================== */
#ifndef DEV_NAME_MAX
#define DEV_NAME_MAX 16  /**< 设备名最长字节数（含结尾 NUL） */
#endif
#ifndef DEV_MAX
#define DEV_MAX 8        /**< 可同时注册的设备数 */
#endif

/* ---------------- 设备表项（注册表与驱动通过它交接） ---------------- */
typedef struct
{
    char        name[DEV_NAME_MAX]; /**< 设备名（注册时拷贝，同名唯一） */
    const void *ops;   /**< 驱动能力函数表（如 uart_ops_t*），由驱动定义并解释 */
    const void *hw;    /**< 平台句柄（设备节点字符串 / HAL 句柄指针），驱动解释 */
    void       *priv;  /**< 驱动私有数据（如已分配的实例句柄），驱动自管，可选 */
    uint8_t     used;  /**< 1=已注册 */
} dev_entry_t;

/* ---------------- 注册表 API（纯登记，不做 I/O） ---------------- */

/**
 * @brief  注册设备（登记即占槽，不打开）
 * @param  name 设备名（同名拒绝；不可为空）
 * @param  ops  驱动能力函数表指针；NULL 也允许（纯登记）
 * @param  hw   平台句柄，原样保存
 * @param  priv 驱动私有数据，原样保存；可选填 NULL，或指向驱动的实例句柄
 * @return 设备表项指针；重名 / 表满 / 参数非法返回 NULL
 */
dev_entry_t *dev_register(const char *name, const void *ops, const void *hw, void *priv);

/**
 * @brief  按名查找设备
 * @param  name 设备名
 * @return 设备表项指针；未注册返回 NULL
 */
dev_entry_t *dev_find(const char *name);

/**
 * @brief  注销设备（回收槽位）
 * @param  name 设备名；未注册的名字调用无副作用
 * @note   只回收注册表槽位。驱动若持有打开的资源，应先走驱动自己的
 *         注销接口关闭（如串口驱动的 uart_dev_unregister）。
 */
void dev_unregister(const char *name);

/** @return 已注册设备数量（供枚举 / 调试界面） */
uint8_t dev_count(void);

/**
 * @brief  按序号取设备名（配合 dev_count 枚举，跳过中间注销留下的空位）
 * @param  seq 序号 0..dev_count()-1
 * @return 设备名；越界返回 NULL
 */
const char *dev_name(uint8_t seq);

#endif /* DEVICE_H */
```

/**
 * @file    device.c
 * @brief   通用接口管理 —— 设备注册表实现（静态表，无动态内存）
 *
 * 对外公开符号：dev_*（device.h 声明，共 5 个）。
 * 文件内私有静态函数：devc_* 前缀。
 */

#include <string.h>
#include "device.h"

static dev_entry_t devs[DEV_MAX];  /* 注册表本体：静态数组，任何平台可编译 */

/* ---------------- 文件内私有工具 ---------------- */

/** 找到同名已注册设备槽号；未注册返回 -1 */
static int devc_slot(const char *name)
{
    int i;

    if (!name)
        return -1;
    for (i = 0; i < DEV_MAX; i++)
        if (devs[i].used && strcmp(devs[i].name, name) == 0)
            return i;
    return -1;
}

/** 找一个空槽；表满返回 -1 */
static int devc_empty(void)
{
    int i;

    for (i = 0; i < DEV_MAX; i++)
        if (!devs[i].used)
            return i;
    return -1;
}

/* ---------------- 注册表 API ---------------- */

dev_entry_t *dev_register(const char *name, const void *ops, const void *hw, void *priv)
{
    int i;

    if (!name || !name[0])
        return NULL;
    if (devc_slot(name) >= 0)   /* 重名拒绝 */
        return NULL;
    i = devc_empty();
    if (i < 0)                  /* 表满 */
        return NULL;

    strncpy(devs[i].name, name, DEV_NAME_MAX - 1);
    devs[i].name[DEV_NAME_MAX - 1] = '\0';  /* 强制收尾 */
    devs[i].ops  = ops;
    devs[i].hw   = hw;
    devs[i].priv = priv;
    devs[i].used = 1;
    return &devs[i];
}

dev_entry_t *dev_find(const char *name)
{
    int i = devc_slot(name);

    return (i < 0) ? NULL : &devs[i];
}

void dev_unregister(const char *name)
{
    int i = devc_slot(name);

    if (i >= 0)
        memset(&devs[i], 0, sizeof(devs[i]));  /* 整槽清零：name/ops/hw/priv/used 一并回收 */
}

uint8_t dev_count(void)
{
    uint8_t n = 0;
    int i;

    for (i = 0; i < DEV_MAX; i++)
        if (devs[i].used)
            n++;
    return n;
}

const char *dev_name(uint8_t seq)
{
    uint8_t seen = 0;
    int i;

    for (i = 0; i < DEV_MAX; i++)
        if (devs[i].used)
        {
            if (seen == seq)
                return devs[i].name;
            seen++;
        }
    return NULL;
}
```

## 通用接口管理（开发的通用层）

公共基础片段：**通用类型定义** + **设备注册表**。串口驱动（串口通信驱动.md）等驱动依赖本片段，两个片段一起导入工程后同目录编译。

### 1. 模块结构

| 文件 | 类型 | 职责 | 平台相关 |
|---|---|---|---|
| common.h | 头文件 | 通用类型定义：① 类型别名、② 波特率常量目录、默认值 | 否 |
| device.h | 头文件 | 设备注册表接口（dev_*）+ ① 配置区（DEV_MAX / DEV_NAME_MAX） | 否 |
| device.c | 通用核心 | 注册表实现：静态表，注册 / 查找 / 注销 / 枚举 | 否 |

> 仅标准 C、零动态内存，任何平台可编译。

### 2. 接口输入 / 输出

| 接口 | 输入（参数） | 输出（返回值） | 说明 |
|---|---|---|---|
| dev_register | name=设备名, ops=驱动能力表(NULL 允许), hw=平台句柄, priv=驱动私有(可选) | **dev_entry_t\***；失败 **NULL**（重名/表满/参数非法） | 登记设备（不打开，不做 I/O） |
| dev_find | name=设备名 | **dev_entry_t\***；未注册 **NULL** | 按名取设备 |
| dev_unregister | name=设备名 | 无 | 注销回收槽位；驱动持有的资源须先自行关闭 |
| dev_count / dev_name | - / seq | 数量 / 名字或 NULL | 枚举设备列表（自动跳过注销空位） |

### 3. 使用示例

```c
#include "device.h"

int main(void)
{
    /* 1) 驱动就绪后登记（示例为通用写法；串口驱动请用其 uart_dev_register） */
    dev_register("uart-gps", NULL, &huart1,        NULL); /* STM32：HAL 句柄 */
    dev_register("uart-dbg", NULL, "/dev/ttyUSB0", NULL); /* Linux：设备节点 */

    /* 2) 业务代码只认名字；ops / hw / priv 由对应驱动解释并操作 */
    dev_entry_t *gps = dev_find("uart-gps");
    if (gps) { /* ... */ }

    /* 3) 注销（纯注册表回收；串口设备用 uart_dev_unregister 先关再注销） */
    dev_unregister("uart-dbg");
    return 0;
}
```

### 4. 配置区

- common.h ① 类型别名：`uart_baud_t` / `uart_len_t`；② 波特率目录 `UART_BAUD_*`（换/加波特率只改这里）；`UART_DEFAULT_BAUD`；
- device.h ① 配置区：`DEV_MAX`=8（可注册设备数）、`DEV_NAME_MAX`=16（名字最长字节数），容量可裁剪。

### 5. 命名规范

| 前缀 | 范围 | 例子 |
|---|---|---|
| dev_ | 注册表公开 API | dev_register、dev_find、dev_count |
| devc_ | device.c 文件内私有静态函数 | devc_slot、devc_empty |

### 6. 说明

- 注册表只做"登记 / 查找 / 注销"，**不做任何 I/O**：ops / hw / priv 原样保存，由驱动定义并解释；
- 同名唯一；注销后同名可重注册；未注册名字的操作无副作用；
- 未来 IIC / SPI 等驱动复用同一注册表：各自实现自己的适配层（参考串口驱动的 uart_dev.c）。

> 串口驱动的完整"注册 → find → 打开 → 收发"流程见 `串口通信驱动.md` 第 10 章。
