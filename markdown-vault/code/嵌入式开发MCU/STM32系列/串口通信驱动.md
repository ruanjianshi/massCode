```c_cpp
#ifndef UART_H
#define UART_H

/**
 * @file    uart.h
 * @brief   通用串口(UART)驱动 —— 接口层 + 统一配置（仅标准 C，无平台依赖）
 *
 * ═══════════ 一、分层设计 ═══════════
 *   应用层（你的 main.c）  只调本文件 4 个接口，不关心底层硬件
 *   通用核心（uart.c）     包装函数，把调用转发到后端操作表 uart_ops_t
 *   后端（二选一参与编译） 实现 uart_default_ops：
 *     - uart_linux.c  嵌入式 Linux（POSIX termios）
 *     - uart_mcu.c    STM32 MCU（HAL 库，换芯片只改配置宏）
 *   换平台 = 换一个后端 .c，应用代码一行不改。
 *
 * ═══════════ 二、本文件 = 唯一的"适配入口" ═══════════
 *   通用配置（① 类型别名 / 默认波特率）与 MCU「硬件适配配置区」（②）
 *   全部集中在本文件：后期换硬件 / 换芯片，只改本文件内的定义即可，
 *   后端 .c 是纯实现，移植时不需要动。
 *
 * ═══════════ 三、命名约定 ═══════════
 *   uart_xxx             公开跨平台 API / 类型（应用层可见）
 *   uart_default_ops     后端必须导出的默认操作表
 *   uart_mcu_xxx         MCU 后端公开符号（如中断入口 uart_mcu_rx_byte）
 *   mcu_xxx / linux_xxx  后端文件内私有静态函数
 *
 * ═══════════ 四、快速上手 ═══════════
 *   uart_t u;
 *   uart_open (&u, "/dev/ttyUSB0", UART_DEFAULT_BAUD); // Linux：设备节点
 *   uart_open (&u, &huart1,        UART_DEFAULT_BAUD); // STM32：HAL 句柄
 *   uart_write(&u, "AT+CGMI\r\n", 9);
 *   int n = uart_read (&u, buf, sizeof(buf));
 *   uart_close(&u);
 *
 *   接口取舍：只保留 open/read/write/close 4 个；
 *   set_config / printf / avail / flush 已在 uart_ops_t 注释中预留。
 */

#include <stddef.h>
#include <stdint.h>

/* ==================== ① 通用配置（类型别名 / 默认值） ==================== */

/** 波特率类型：想改宽度（如 16 位）只改这一行 */
typedef uint32_t uart_baud_t;

/** 收发长度类型：想改宽度（如 32 位）只改这一行 */
typedef uint16_t uart_len_t;

/* ---- 标准波特率目录（换/加波特率只改这里，后端查表自动覆盖新值） ----
 * 典型嵌入式写法：朴素 #define 常量，数值即波特率，可 #if 判断；
 * 0 为特殊值（对应 termios B0，不通信）。
 * 高速档（460800 及以上）：仅 Linux termios 有 Bxxx 常量，
 * macOS / 其他 POSIX 无定义，后端引用时用 #ifdef __linux__ 保护。 */
#define UART_BAUD_0        0
#define UART_BAUD_50       50
#define UART_BAUD_75       75
#define UART_BAUD_110      110
#define UART_BAUD_134      134
#define UART_BAUD_150      150
#define UART_BAUD_200      200
#define UART_BAUD_300      300
#define UART_BAUD_600      600
#define UART_BAUD_1200     1200
#define UART_BAUD_1800     1800
#define UART_BAUD_2400     2400
#define UART_BAUD_4800     4800
#define UART_BAUD_9600     9600
#define UART_BAUD_19200    19200
#define UART_BAUD_38400    38400
#define UART_BAUD_57600    57600
#define UART_BAUD_115200   115200
#define UART_BAUD_230400   230400
/* 高速档：仅 Linux termios 有 Bxxx 常量 */
#define UART_BAUD_460800   460800
#define UART_BAUD_500000   500000
#define UART_BAUD_576000   576000
#define UART_BAUD_921600   921600
#define UART_BAUD_1000000  1000000
#define UART_BAUD_1152000  1152000
#define UART_BAUD_1500000  1500000
#define UART_BAUD_2000000  2000000
#define UART_BAUD_2500000  2500000
#define UART_BAUD_3000000  3000000
#define UART_BAUD_3500000  3500000
#define UART_BAUD_4000000  4000000

/** 默认波特率（应用层可直接作为 uart_open 的 baud 参数） */
#define UART_DEFAULT_BAUD UART_BAUD_115200

/* ------------------------------------------------------------------ */
/* 句柄类型：应用层只持有指针，内部实例由各后端自行定义（不透明封装）     */
/* ------------------------------------------------------------------ */
typedef struct uart uart_t;

/* ------------------------------------------------------------------ */
/* 后端操作表：每个后端 .c 实现以下函数指针并导出 uart_default_ops        */
/* ------------------------------------------------------------------ */
typedef struct {
    /** 打开串口：0=成功，-1=失败 */
    int (*open)(uart_t *u, const void *hw, uart_baud_t baud);

    /** 发送 len 字节：返回实际发送字节数，失败返回 -1 */
    int (*write)(uart_t *u, const void *buf, uart_len_t len);

    /** 非阻塞接收：返回实际接收字节数（无数据返回 0），失败返回 -1 */
    int (*read)(uart_t *u, void *buf, uart_len_t len);

    /** 关闭串口并释放资源（无返回值） */
    void (*close)(uart_t *u);

    /* ============ 预留扩展（按需实现后在此追加函数指针） ============
     * int (*set_config)(uart_t*, uint8_t data_bits, char parity,
     *                   uint8_t stop_bits, uint8_t blocking);  数据位/校验/停止位/阻塞
     * int (*printf)(uart_t*, const char *fmt, ...);            格式化发送
     * int (*avail)(uart_t*);                                   接收缓冲可读字节数
     * int (*flush)(uart_t*);                                   清空接收缓冲
     * ============================================================== */
} uart_ops_t;

/* 应用层句柄的实际结构 */
struct uart {
    const uart_ops_t *ops; /* 后端操作表（open 时绑定） */
    void *hw;              /* 平台句柄：Linux=文件描述符；STM32=HAL 句柄指针 */
};

/* 后端导出的默认操作表（uart_linux.c 或 uart_mcu.c 中定义其一） */
extern const uart_ops_t uart_default_ops;

/* ------------------------------------------------------------------ */
/* 应用层统一接口（仅此 4 个，所有平台用法一致）                          */
/* ------------------------------------------------------------------ */

/**
 * @brief  打开串口（默认 8N1、非阻塞读）
 * @param  u    应用层句柄（由调用方提供存储空间）
 * @param  hw   平台句柄：
 *              - Linux：设备节点字符串，如 "/dev/ttyUSB0"；
 *              - STM32：HAL 句柄指针，如 &huart1
 * @param  baud 波特率（可填 UART_DEFAULT_BAUD 宏），如 9600 / 115200 / 230400
 * @return 0=成功；-1=失败
 */
int uart_open(uart_t *u, const void *hw, uart_baud_t baud);

/**
 * @brief  指定后端打开串口（进阶接口，设备管理等按设备选后端时使用）
 * @param  u    句柄
 * @param  ops  后端操作表（如 &uart_default_ops，或设备表里登记的定制后端）
 * @param  hw   平台句柄（同 uart_open）
 * @param  baud 波特率（可填 UART_DEFAULT_BAUD 宏）
 * @return 0=成功；-1=失败
 * @note   uart_open 等价于 uart_open_ops(u, &uart_default_ops, hw, baud)
 */
int uart_open_ops(uart_t *u, const uart_ops_t *ops, const void *hw, uart_baud_t baud);

/**
 * @brief  发送数据（阻塞，全部发出或超时）
 * @param  u   句柄
 * @param  buf 数据缓冲区
 * @param  len 字节数
 * @return 实际发送字节数；失败返回 -1
 */
int uart_write(uart_t *u, const void *buf, uart_len_t len);

/**
 * @brief  接收数据（非阻塞：读空即返回）
 * @param  u   句柄
 * @param  buf 接收缓冲区
 * @param  len 缓冲区容量（字节）
 * @return 实际接收字节数（无数据时 0）；失败返回 -1
 */
int uart_read(uart_t *u, void *buf, uart_len_t len);

/**
 * @brief  关闭串口，释放资源
 * @param  u 句柄（关闭后仍可再次 open）
 */
void uart_close(uart_t *u);

/* ==================== ② MCU 硬件适配配置区 ★换硬件只改这里★ ====================
 *
 * 以下宏**只在链接 MCU 后端（uart_mcu.c）时生效**；Linux 后端不展开这些
 * 宏，可安全共存。uart_mcu.c 会通过 UART_MCU_HAL_HEADER 引入厂商头文件，
 * 本文件不 include 任何厂商头（保持跨平台）。
 *
 * 换硬件时按下列顺序逐条改，改完只重编译，源码不动。
 */

/* --- 2.1 芯片 HAL 头文件：默认 STM32F1；换芯片只改这一行 ---
 * 例：F4 → "stm32f4xx_hal.h"；G0 → "stm32g0xx_hal.h"；L4 → "stm32l4xx_hal.h" */
#ifndef UART_MCU_HAL_HEADER
#define UART_MCU_HAL_HEADER "stm32f1xx_hal.h"
#endif

/* --- 2.2 外设句柄类型：换芯片类型只改这一行（GD32、NXP 等换成自己的句柄类型） --- */

/* --- 2.3 串口引脚：默认 USART1 = PA9(TX) / PA10(RX)；换板子只改这里 --- */

## Fragment: uart.c
```c_cpp
/**
 * @file    uart.c
 * @brief   串口驱动 · 通用核心（跨平台包装层，任何平台原样编译）
 *
 * 职责：
 *   1. uart_open() 时把编译期选定的后端操作表 uart_default_ops 绑定到句柄；
 *   2. 其余接口做空指针防护后，把调用转发给操作表对应函数指针。
 * 本文件不包含任何平台代码。
 */
#include "uart.h"

#include <string.h> /* memset */

int uart_open_ops(uart_t *u, const uart_ops_t *ops, const void *hw, uart_baud_t baud)
{
    /* 参数校验：句柄 / 后端 / 硬件句柄为空直接失败，避免后续空指针解引用 */
    if (!u || !ops || !ops->open || !hw)
        return -1;

    /* 先清零再绑定：
     * - 保证 ops / hw 初始值确定；
     * - 重复 open（未 close）时不残留上一次的旧状态。 */
    memset(u, 0, sizeof(*u));
    u->ops = ops; /* 绑定调用方指定的后端（设备管理等场景可按设备选后端） */

    return ops->open(u, hw, baud);
}

int uart_open(uart_t *u, const void *hw, uart_baud_t baud)
{
    /* 便捷封装：绑定链接期默认后端（uart_default_ops） */
    return uart_open_ops(u, &uart_default_ops, hw, baud);
}

int uart_write(uart_t *u, const void *buf, uart_len_t len)
{
    /* 未 open（ops 为空）、后端未实现、或 len>0 却无缓冲区 → 保守返回 -1 */
    if (!u || !u->ops || !u->ops->write || (len > 0 && !buf))
        return -1;

    return u->ops->write(u, buf, len);
}

int uart_read(uart_t *u, void *buf, uart_len_t len)
{
    if (!u || !u->ops || !u->ops->read || (len > 0 && !buf))
        return -1;

    return u->ops->read(u, buf, len);
}

void uart_close(uart_t *u)
{
    if (u && u->ops && u->ops->close)
        u->ops->close(u);
}

## Fragment: uart_linux.c
```c_cpp
/**
 * @file    uart_linux.c
 * @brief   串口驱动 · 嵌入式 Linux 后端（POSIX termios）
 *
 * 实现 uart_default_ops；应用层代码见 uart.h，无需改动。
 *
 * 特性：
 *   - 默认 8N1、原始模式（无回显 / 无行缓冲 / 无信号产生）；
 *   - 非阻塞读：VMIN=0 / VTIME=1，最多等 0.1s，无数据立即返回 0；
 *   - 接收缓冲由内核 tty 层提供（flip buffer + 行规程），驱动层无需自建环形缓冲；
 *   - 设备节点：/dev/ttySx（片上 UART）、/dev/ttyUSBx（USB 转串）、
 *               /dev/ttyAMAx、/dev/ttyTHSx 等；
 *   - 依赖：仅标准 C + POSIX（open / termios / read / write）。
 *
 * 编译：gcc uart.c uart_linux.c main.c -o app
 *
 * 对外公开符号：仅 uart_default_ops（uart.h 声明）；其余符号一律 static。
 */
#include "uart.h"

/* 平台特性宏：Linux 需完整 termios 特性；macOS 需 BSD 定义 */
#if defined(__linux__)
#define _GNU_SOURCE
#elif defined(__APPLE__)
#define _DARWIN_C_SOURCE
#endif

#include <errno.h>
#include <fcntl.h>
#include <stdint.h>
#include <termios.h>
#include <unistd.h>

/**
 * @brief  波特率数值 → termios 常量
 * @param  baud 波特率，如 115200
 * @return termios 的 Bxxx 常量；不支持时返回 B0 并置 errno=EINVAL
 * @note   表很小（≤31 项），线性查表即可；Bxxx 是 POSIX 专有常量：
 *         B460800 及以上仅 Linux 定义（macOS/POSIX 无），故高速档条目用
 *         #ifdef __linux__ 包含。换/加波特率：先在 uart.h 目录加 #define，
 *         再在下面表里加一行。
 */
/* 波特率映射表：数值 → termios Bxxx 常量（嵌入式常规静态常量查表，直观可调试） */
static const struct
{
    uart_baud_t num;  /* uart.h 目录中的波特率数值 */
    speed_t     spd;  /* termios 常量 */
} baud_map[] =
{
    { UART_BAUD_0,       B0 },       /* 0：特殊值（不通信） */
    { UART_BAUD_50,      B50 },
    { UART_BAUD_75,      B75 },
    { UART_BAUD_110,     B110 },
    { UART_BAUD_134,     B134 },
    { UART_BAUD_150,     B150 },
    { UART_BAUD_200,     B200 },
    { UART_BAUD_300,     B300 },
    { UART_BAUD_600,     B600 },
    { UART_BAUD_1200,    B1200 },
    { UART_BAUD_1800,    B1800 },
    { UART_BAUD_2400,    B2400 },
    { UART_BAUD_4800,    B4800 },
    { UART_BAUD_9600,    B9600 },
    { UART_BAUD_19200,   B19200 },
    { UART_BAUD_38400,   B38400 },
    { UART_BAUD_57600,   B57600 },
    { UART_BAUD_115200,  B115200 },
    { UART_BAUD_230400,  B230400 },
#ifdef __linux__  /* B460800 及以上为 Linux 专有（macOS/POSIX 无 Bxxx 常量） */
    { UART_BAUD_460800,  B460800 },
    { UART_BAUD_500000,  B500000 },
    { UART_BAUD_576000,  B576000 },
    { UART_BAUD_921600,  B921600 },
    { UART_BAUD_1000000, B1000000 },
    { UART_BAUD_1152000, B1152000 },
    { UART_BAUD_1500000, B1500000 },
    { UART_BAUD_2000000, B2000000 },
    { UART_BAUD_2500000, B2500000 },
    { UART_BAUD_3000000, B3000000 },
    { UART_BAUD_3500000, B3500000 },
    { UART_BAUD_4000000, B4000000 },
#endif
};

static speed_t linux_baud_to_speed(uart_baud_t baud)
{
    size_t i;

    for (i = 0; i < sizeof(baud_map) / sizeof(baud_map[0]); i++)
    {
        if (baud_map[i].num == baud)
            return baud_map[i].spd;
    }

    errno = EINVAL;
    return B0;
}

/**
 * @brief  打开串口设备并配置为 8N1 原始模式
 * @param  u    句柄
 * @param  hw   设备节点路径字符串，如 "/dev/ttyUSB0"
 * @param  baud 波特率
 * @return 0=成功；-1=失败（errno 说明原因）
 */
static int linux_open(uart_t *u, const void *hw, uart_baud_t baud)
{
    const char *dev = (const char *)hw;

    if (!dev)
    {
        errno = EINVAL;
        return -1;
    }

    /* 1) 打开设备节点：
     *    O_RDWR    读写；
     *    O_NOCTTY  不把设备当作控制终端（防止被键盘信号干扰）；
     *    O_NDELAY  打开时不等待 DCD 载波（避免挂起）。 */
    int fd = open(dev, O_RDWR | O_NOCTTY | O_NDELAY);
    if (fd < 0)
        return -1;

    /* 2) 恢复阻塞模式：
     *    打开参数里的 O_NDELAY 只是为了不等 DCD；真正的读/写行为
     *    由 termios 的 VMIN/VTIME 决定（见第 6 步），所以这里把它清掉。 */
    if (fcntl(fd, F_SETFL, 0) < 0)
    {
        close(fd);
        return -1;
    }

    /* 3) 读取当前 termios 配置，在其基础上修改 */
    struct termios t;
    if (tcgetattr(fd, &t) != 0)
    {
        close(fd);
        return -1;
    }

    /* 4) 原始模式：一次关掉回显(ECHO)、行缓冲(ICANON)、
     *    信号生成(ISIG)、软流控(IXON/IXOFF) 等所有"终端化"行为 */
    cfmakeraw(&t);

    /* 5) 固定 8N1、关闭硬件流控 */
    t.c_cflag &= ~(CSIZE | PARENB | PARODD | CSTOPB | CRTSCTS);
    t.c_cflag |= CS8;

    /* 波特率：不支持的值直接失败 */
    speed_t spd = linux_baud_to_speed(baud);
    if (spd == B0)
    {
        close(fd);
        return -1;
    }
    if (cfsetispeed(&t, spd) || cfsetospeed(&t, spd))
    {
        close(fd);
        return -1;
    }

    /* 6) 非阻塞读：VMIN=0（不要求凑满字节数）+ VTIME=1（最多等 0.1s），
     *    之后 read() 无数据时返回 0，不会卡死主循环 */
    t.c_cc[VMIN] = 0;
    t.c_cc[VTIME] = 1;

    /* 7) TCSANOW 立即生效，并清空收发缓冲里的旧数据 */
    if (tcsetattr(fd, TCSANOW, &t) != 0)
    {
        close(fd);
        return -1;
    }
    tcflush(fd, TCIOFLUSH);

    /* 8) 文件描述符存入句柄（void* 容纳 int 的保证来自 intptr_t） */
    u->hw = (void *)(intptr_t)fd;
    return 0;
}

/**
 * @brief  发送 len 字节
 * @return 实际写入字节数（通常等于 len）；失败返回 -1
 */
static int linux_write(uart_t *u, const void *buf, uart_len_t len)
{
    return (int)write((int)(intptr_t)u->hw, buf, len);
}

/**
 * @brief  非阻塞接收
 * @return 实际读取字节数（无数据返回 0）；失败返回 -1
 */
static int linux_read(uart_t *u, void *buf, uart_len_t len)
{
    return (int)read((int)(intptr_t)u->hw, buf, len);
}

/**
 * @brief  关闭设备节点并置空句柄
 */
static void linux_close(uart_t *u)
{
    if (u->hw)
    {
        close((int)(intptr_t)u->hw);
        u->hw = NULL;
    }
}

    .open  = linux_open,
    .write = linux_write,
    .read  = linux_read,
    .close = linux_close,

/**
 * @file    uart_mcu.c
 * @brief   串口驱动 · MCU 后端（HAL 库，跨芯片可移植版）—— 纯实现
 *
 * ═══════════ 移植原则 ═══════════
 * 所有"换硬件"的开关（芯片 HAL 头 / 句柄类型 / 引脚 / 时钟使能 /
 * 复用功能号 / 实例数 / 环形缓冲大小）都集中在 uart.h 的
 * 「② MCU 硬件适配配置区」。本文件是纯实现，**移植时不需要改动**。
 *
 * ═══════════ 中断接收数据流 ═══════════
 *   外部数据 → USART RX 中断
 *     → HAL_UART_Receive_IT() 每收 1 字节触发 HAL_UART_RxCpltCallback()
 *     → uart_mcu_rx_byte() → mcu_rx_push() 压入环形缓冲 rx[]
 *   应用层 uart_read() → mcu_read() → mcu_rx_pop() 从缓冲取出数据
 *   （其他芯片 / 裸机：在 ISR 里直接调 uart_mcu_rx_byte(&hw, byte) 即可，
 *     同样走上面的环形缓冲，不用改本文件源码）
 *
 * ═══════════ 使用 ═══════════
 *   uart_open(&u, &huart1, UART_DEFAULT_BAUD); // huart1 来自 CubeMX
 *   CubeMX 里开对应 USART 的全局中断(NVIC)；GPIO 由本文件自动初始化。
 *
 * ═══════════ 对外公开符号 ═══════════
 *   uart_default_ops   后端操作表（链接期唯一入口，uart.h 声明）
 *   uart_mcu_rx_byte   跨芯片中断入口（uart.h 声明）
 *   其余符号一律 static，仅本文件可见。
 */
#include "uart.h"

/* HAL 头文件：由 uart.h 的 UART_MCU_HAL_HEADER 宏指定（换芯片只改宏） */
#include UART_MCU_HAL_HEADER

/* ==================== 通用实现区（移植无需改动） ==================== */

/* ---------------- 环形缓冲 + 实例表（中断与主循环共享） ---------------- */
typedef struct {
    UART_MCU_HANDLE_T *huart;      /* 绑定的外设句柄（区分实例用） */
    uint8_t rx[UART_RX_BUF_SIZE];  /* 环形缓冲（大小见 uart.h 2.6） */
    volatile uint16_t head;        /* 写指针：ISR 压入后前进 */
    volatile uint16_t tail;        /* 读指针：主循环取出后前进 */
    uint8_t used;                  /* 槽位是否已分配 */
} uart_inst_t;

/* 实例表：静态数组，避免动态内存分配（嵌入式环境安全） */
static uart_inst_t insts[UART_MCU_MAX];

/**
 * @brief  按外设句柄查找已分配的实例
 * @param  hw 外设句柄
 * @return 实例指针；未找到返回 NULL
 */
static uart_inst_t *mcu_inst_find(const void *hw)
{
    for (int i = 0; i < UART_MCU_MAX; i++)
    {
        if (insts[i].used && insts[i].huart == hw)
            return &insts[i];
    }
    return NULL;
}

/**
 * @brief  分配一个空闲实例槽位
 * @return 实例指针；已满（同时使用超过 UART_MCU_MAX 路串口）返回 NULL
 */
static uart_inst_t *mcu_inst_alloc(void)
{
    for (int i = 0; i < UART_MCU_MAX; i++)
    {
        if (!insts[i].used)
        {
            insts[i].used = 1;
            insts[i].head = insts[i].tail = 0; /* 清空缓冲 */
            return &insts[i];
        }
    }
    return NULL;
}

/**
 * @brief  向环形缓冲压入 1 字节（仅中断上下文调用）
 * @param  i 实例
 * @param  b 字节
 * @note   单生产者模型：生产端只有中断这一个入口；
 *         容量 = SIZE-1；head 追上 tail 判定为满，满则丢弃新字节
 *         （宁可丢新，也不覆盖尚未被读走的数据）。
 */
static void mcu_rx_push(uart_inst_t *i, uint8_t b)
{
    uint16_t nxt = (i->head + 1) & (UART_RX_BUF_SIZE - 1); /* 2 的幂：位与 = 取模 */
    if (nxt != i->tail)
    {
        i->rx[i->head] = b;
        i->head = nxt;
    }
}

/**
 * @brief  从环形缓冲取出 1 字节（主循环调用）
 * @param  i 实例
 * @param  b 输出：取出的字节
 * @return 1=取到；0=缓冲空
 */
static int mcu_rx_pop(uart_inst_t *i, uint8_t *b)
{
    if (i->head == i->tail) /* 空 */
        return 0;

    *b = i->rx[i->tail];
    i->tail = (i->tail + 1) & (UART_RX_BUF_SIZE - 1);
    return 1;
}

/* ---------------- 硬件初始化（全部由 uart.h 配置区宏驱动） ---------------- */

/**
 * @brief  初始化串口硬件：开外设时钟 + 开 GPIO 时钟 + TX/RX 引脚复用
 * @note   幂等：CubeMX 已初始化也无副作用；裸机工程同样自包含。
 *         所有引脚/时钟/复用号来自 uart.h ② 硬件适配配置区。
 */
static void mcu_hw_init(void)
{
    UART_MCU_CLK_ENABLE();      /* 外设时钟：换芯片 / 时钟线只改 uart.h */
    UART_MCU_GPIO_CLK_ENABLE(); /* GPIO 时钟：换端口时同步改 uart.h */

    /* TX/RX 引脚：复用推挽输出、无上下拉、高速 */
    GPIO_InitTypeDef g;
    g.Pin = UART_MCU_TX_PIN;
    g.Mode = GPIO_MODE_AF_PP;
    g.Pull = GPIO_NOPULL;
    g.Speed = GPIO_SPEED_FREQ_HIGH;
#if UART_MCU_AF
    g.Alternate = UART_MCU_AF; /* F4/H7 等：写复用功能号；F1 无该成员，AF=0 整段被裁掉 */
#endif
    HAL_GPIO_Init(UART_MCU_TX_PORT, &g);

    g.Pin = UART_MCU_RX_PIN;
    HAL_GPIO_Init(UART_MCU_RX_PORT, &g);
}

/* ---------------- 后端 4 接口 ---------------- */

/**
 * @brief  打开串口：初始化硬件 + 配置 8N1 + 启动中断接收
 * @param  u    句柄
 * @param  hw   HAL 句柄指针，如 &huart1
 * @param  baud 波特率
 * @return 0=成功；-1=失败（实例满 / HAL_UART_Init 失败）
 */
static int mcu_open(uart_t *u, const void *hw, uart_baud_t baud)
{
    UART_MCU_HANDLE_T *huart = (UART_MCU_HANDLE_T *)hw;

    if (!huart)
        return -1;

    /* 分配实例槽（每路串口一个，上限 UART_MCU_MAX，见 uart.h 2.6） */
    uart_inst_t *i = mcu_inst_alloc();
    if (!i)
        return -1;
    i->huart = huart;

    /* 先开时钟 + 配 GPIO（宏驱动，幂等）；再固定 8N1、无流控、16 倍过采样 */
    mcu_hw_init();
    huart->Init.BaudRate = baud;
    huart->Init.WordLength = UART_WORDLENGTH_8B;
    huart->Init.Parity = UART_PARITY_NONE;
    huart->Init.StopBits = UART_STOPBITS_1;
    huart->Init.Mode = UART_MODE_TX_RX;
    huart->Init.HwFlowCtl = UART_HWCONTROL_NONE;
    huart->Init.OverSampling = UART_OVERSAMPLING_16;
    if (HAL_UART_Init(huart) != HAL_OK)
    {
        i->used = 0; /* 初始化失败：归还实例槽 */
        return -1;
    }

    /* 启动"每收 1 字节进环形缓冲"的中断接收（数据流见文件头注释）；
     * 失败同样归还槽位，保证 open 失败后句柄干净 */
    if (HAL_UART_Receive_IT(huart, (uint8_t *)&huart->Instance->DR, 1) != HAL_OK)
    {
        i->used = 0;
        return -1;
    }

    u->hw = (void *)hw; /* 全部成功后才绑定句柄，不留半初始化状态 */
    return 0;
}

/**
 * @brief  发送 len 字节（阻塞，最多等 1000ms）
 * @return 实际发送字节数；失败返回 -1
 */
static int mcu_write(uart_t *u, const void *buf, uart_len_t len)
{
    return HAL_UART_Transmit((UART_MCU_HANDLE_T *)u->hw, (const uint8_t *)buf, len, 1000) == HAL_OK
               ? (int)len
               : -1;
}

/**
 * @brief  非阻塞读环形缓冲
 * @return 实际读出字节数（无数据返回 0）；实例不存在返回 -1
 */
static int mcu_read(uart_t *u, void *buf, uart_len_t len)
{
    uart_inst_t *i = mcu_inst_find(u->hw);
    if (!i)
        return -1;

    uint8_t *p = (uint8_t *)buf;
    uart_len_t n = 0;
    while (n < len) /* 读满或读空即停（非阻塞） */
    {
        uint8_t b;
        if (!mcu_rx_pop(i, &b))
            break;
        p[n++] = b;
    }
    return n;
}

/**
 * @brief  关闭：停止中断接收，归还实例槽
 */
static void mcu_close(uart_t *u)
{
    uart_inst_t *i = mcu_inst_find(u->hw);
    if (i)
    {
        HAL_UART_AbortReceive((UART_MCU_HANDLE_T *)u->hw); /* 停止中断接收 */
        i->used = 0;                                       /* 归还实例槽 */
    }
    u->hw = NULL;
}

/* ---------------- 中断接收入口（与芯片无关） ---------------- */

/**
 * @brief  跨芯片通用中断入口：收到 1 字节 → 压入对应实例的环形缓冲
 * @param  hw 外设句柄（须先 open 过）
 * @param  b  收到的字节
 * @note   任何芯片的 ISR 都可直接调用；STM32 的 HAL 回调内部也是调它。
 */
void uart_mcu_rx_byte(const void *hw, uint8_t b)
{
    uart_inst_t *i = mcu_inst_find(hw);
    if (i)
        mcu_rx_push(i, b);
}

/**
 * @brief  STM32 HAL 接收完成回调（CubeMX 全局中断自动走到这里）
 * @note   动作：字节入缓冲 + 重新挂起下一个字节的中断接收。
 *         F1/F4/L4… 各系列 HAL 签名一致，本文件通吃。
 */
void HAL_UART_RxCpltCallback(UART_HandleTypeDef *huart)
{
    uart_mcu_rx_byte((const void *)huart, (uint8_t)(huart->Instance->DR & 0xFF));
    HAL_UART_Receive_IT(huart, (uint8_t *)&huart->Instance->DR, 1); /* 继续收下一个 */
}

/* 默认操作表：本后端在链接期的唯一公开符号 */
const uart_ops_t uart_default_ops = {
    .open  = mcu_open,
    .write = mcu_write,
    .read  = mcu_read,
    .close = mcu_close,
};

/**
 * @file    device.c
 * @brief   设备管理表实现 —— 静态数组，无动态内存分配
 *
 * 对外公开符号：dev_*（device.h 声明，共 10 个）；
 * 文件内私有：devc_* 均 static。
 */
#include "device.h"

/* 设备表：静态数组（嵌入式惯例，避免堆） */
static dev_entry_t devs[DEV_MAX];

/**
 * @brief  按名字找表项下标
 * @param  name 设备名
 * @return 表项下标；未注册返回 -1
 */
static int devc_slot(const char *name)
{
    for (int i = 0; i < DEV_MAX; i++)
    {
        if (devs[i].used && strcmp(devs[i].name, name) == 0)
            return i;
    }
    return -1;
}

/**
 * @brief  找一个空槽位
 * @return 空槽下标；表满返回 -1
 */
static int devc_empty(void)
{
    for (int i = 0; i < DEV_MAX; i++)
    {
        if (!devs[i].used)
            return i;
    }
    return -1;
}

dev_entry_t *dev_register(const char *name, const uart_ops_t *ops, const void *hw)
{
    if (!name || !name[0] || !hw)
        return NULL;                 /* 参数非法 */
    if (devc_slot(name) >= 0)
        return NULL;                 /* 重名拒绝 */

    int i = devc_empty();
    if (i < 0)
        return NULL;                 /* 表满 */

    dev_entry_t *d = &devs[i];
    d->ops = ops ? ops : &uart_default_ops; /* NULL = 链接期默认后端 */
    d->hw  = hw;

    /* 安全拷贝名字：取前 N-1 字节，最后一位强制 '\0' */
    strncpy(d->name, name, DEV_NAME_MAX - 1);
    d->name[DEV_NAME_MAX - 1] = '\0';

    d->used = 1;
    return d;
}

dev_entry_t *dev_find(const char *name)
{
    int i = (name && name[0]) ? devc_slot(name) : -1;
    return (i >= 0) ? &devs[i] : NULL;
}

void dev_unregister(const char *name)
{
    int i = (name && name[0]) ? devc_slot(name) : -1;
    if (i < 0)
        return;

    dev_entry_t *d = &devs[i];
    if (d->u.ops)
        uart_close(&d->u);           /* 若已打开先关闭 */
    memset(d, 0, sizeof(*d));        /* 整槽清零（含 used=0） */
}

int dev_open(dev_entry_t *d, uart_baud_t baud)
{
    if (!d || !d->used)
        return -1;
    return uart_open_ops(&d->u, d->ops, d->hw, baud); /* 用该设备自己的后端打开 */
}

int dev_write(dev_entry_t *d, const void *buf, uart_len_t len)
{
    if (!d || !d->used)
        return -1;
    return uart_write(&d->u, buf, len);
}

int dev_read(dev_entry_t *d, void *buf, uart_len_t len)
{
    if (!d || !d->used)
        return -1;
    return uart_read(&d->u, buf, len);
}

void dev_close(dev_entry_t *d)
{
    if (d && d->used && d->u.ops)
        uart_close(&d->u);
}

uint8_t dev_count(void)
{
    uint8_t n = 0;
    for (int i = 0; i < DEV_MAX; i++)
        n += devs[i].used ? 1u : 0u;
    return n;
}

const char *dev_name(uint8_t i)
{
    return (i < DEV_MAX && devs[i].used) ? devs[i].name : NULL;
}

# 通用串口(UART)驱动 —— 跨平台可移植代码模块

## 1. 模块结构（文件分段）

| 文件 | 类型 | 职责 | 平台相关 |
|---|---|---|---|
| uart.h | 头文件 | 公开接口 + 类型 + **①通用配置、②MCU 硬件适配配置区（换硬件只改这一个文件）** + 命名约定 | 否 |
| uart.c | 通用核心 | API 包装：绑定后端 → 转发到 ops 函数表 | 否 |
| device.h | 头文件 | 设备管理接口（dev_*，见第 10 章）+ ① 配置区（DEV_MAX / DEV_NAME_MAX） | 否 |
| device.c | 通用核心 | 设备注册表实现：静态表，注册 / 查找 / 注销 | 否 |
| uart_linux.c | 后端 | termios 实现，导出 `uart_default_ops` | Linux |
| uart_mcu.c | 后端 | HAL 实现 + 环形缓冲，导出 `uart_default_ops` | STM32 |

> 后端**二选一**参与编译；应用代码含 `"uart.h"` 直接用，或含 `"device.h"`（内含 uart.h）走设备管理。

## 2. 接口说明（输入 / 输出）

### 2.1 应用层统一接口

| 接口 | 输入（参数） | 输出（返回值） | 说明 |
|---|---|---|---|
| uart_open | u=句柄, hw=平台句柄, baud=波特率（可填 UART_DEFAULT_BAUD） | **0**=成功；**-1**=失败 | 默认 8N1、非阻塞接收；等价 uart_open_ops(默认后端) |
| uart_open_ops | u, ops=后端表, hw, baud（进阶） | 同 uart_open | 指定后端打开（设备管理内部使用，见第 10 章） |
| uart_write | u, buf=发送数据, len=字节数 | **实际发送字节数**；**-1**=失败 | 阻塞发送 |
| uart_read | u, buf=接收区, len=容量 | **实际接收字节数**；**0**=暂无数据；**-1**=失败 | 非阻塞 |
| uart_close | u | 无 | 释放资源，可再次 open |

```c
/* 函数签名（uart.h） */
int  uart_open    (uart_t *u, const void *hw, uart_baud_t baud);
int  uart_open_ops(uart_t *u, const uart_ops_t *ops, const void *hw, uart_baud_t baud); /* 进阶 */
int  uart_write   (uart_t *u, const void *buf, uart_len_t len);
int  uart_read    (uart_t *u, void *buf, uart_len_t len);
void uart_close   (uart_t *u);
```

> 波特率目录在 uart.h「① 通用配置」（朴素 `#define` 常量，常用如 `UART_BAUD_115200` 即 `UART_DEFAULT_BAUD`）；Linux 后端用一张静态常量表把数值映射成 termios `Bxxx`（嵌入式常规查表写法）。**加自定义波特率 = uart.h 目录加一行 `#define` + 后端表加一行**。

### 2.2 后端需要提供的符号

| 符号 | 说明 |
|---|---|
| uart_ops_t | 后端操作表：4 个函数指针（open/write/read/close）；预留 set_config / printf / avail / flush |
| uart_default_ops | 后端导出的默认操作表，uart.c 通过它绑定实现 |
| uart_mcu_rx_byte() | MCU 中断接收入口（其他芯片 / 裸机 ISR 直接调用） |

## 3. 输出（返回值）约定

| 返回值 | 含义 | 适用 |
|---|---|---|
| 0 | 操作成功 | uart_open |
| \>0 | 实际收发字节数 | uart_write / uart_read |
| 0 | 暂无数据（非阻塞，**不是错误**） | uart_read |
| -1 | 失败（Linux 置 errno；MCU 由 HAL 状态决定） | 全部 |

## 4. 快速上手（完整代码块）

### 4.1 Linux（编译运行）

```bash
gcc uart.c uart_linux.c main.c -o app
./app
```

```c
#include "uart.h"

int main(void)
{
    uart_t u;
    char buf[64];

    if (uart_open(&u, "/dev/ttyUSB0", UART_DEFAULT_BAUD) != 0)
        return -1;                    /* 打开失败 */

    uart_write(&u, "AT+CGMI\r\n", 9);
    int n = uart_read(&u, buf, sizeof(buf));  /* n: 实际收到的字节数 */

    uart_close(&u);
    return 0;
}
```

### 4.2 STM32（CubeMX / CubeIDE）

工程加 `uart.c + uart_mcu.c`，CubeMX 开 USART1 + 全局中断(NVIC) 即可（**GPIO 自动初始化**，见 5.2）。应用代码与 Linux 完全一致：

```c
#include "uart.h"

int main(void)
{
    uart_t u;

    if (uart_open(&u, &huart1, UART_DEFAULT_BAUD) != 0)  /* huart1 来自 CubeMX */
        return -1;

    uart_write(&u, "AT+CGMI\r\n", 9);
    char rx[64];
    int n = uart_read(&u, rx, sizeof(rx));    /* 注意：非阻塞，无数据时 n=0 */

    uart_close(&u);
    return 0;
}
```

## 5. 移植指南（分场景，每段说清"改哪里"）

> 所有适配开关集中在 uart.h 的「② MCU 硬件适配配置区」；下面的代码块就是"改哪里"的示例。

### 5.1 换平台：Linux ↔ STM32

接口不变、应用代码**一行不改**；只换参与编译的后端（.c）和 `uart_open` 第二个参数（设备节点 ↔ HAL 句柄）。

### 5.2 换芯片系列（F1 → F4）：只改 2 处宏

```c
/* uart.h ② 硬件适配配置区 */
#define UART_MCU_HAL_HEADER "stm32f4xx_hal.h"  /* 1) 芯片 HAL 头文件 */
#define UART_MCU_AF         GPIO_AF7_USART1    /* 2) 复用功能号 */
```

### 5.3 换板子（换引脚）：只改引脚 + GPIO 时钟宏

```c
#define UART_MCU_TX_PORT GPIOB   /* 例：PB6 / PB7 用 USART1 复用 */
#define UART_MCU_TX_PIN  GPIO_PIN_6
#define UART_MCU_RX_PORT GPIOB
#define UART_MCU_RX_PIN  GPIO_PIN_7
#define UART_MCU_GPIO_CLK_ENABLE() __HAL_RCC_GPIOB_CLK_ENABLE()  /* 端口变了要同步 */
```

### 5.4 换芯片厂商（GD32 为例）：改类型 + 时钟宏 + 中断入口

```c
#define UART_MCU_HAL_HEADER    "gd32f1x0_usart.h"
#define UART_MCU_HANDLE_T      usart_typedef              /* 句柄类型一行替换 */
#define UART_MCU_CLK_ENABLE()      rcu_periph_clock_enable(RCU_USART1)
#define UART_MCU_GPIO_CLK_ENABLE() rcu_periph_clock_enable(RCU_GPIOA)
/* ISR 里：uart_mcu_rx_byte(&usart1, byte); —— 源码不改 */
```

## 6. 接收机制（平台差异）

接收路径按平台区分：

- **Linux：无需自建缓冲**。内核 tty 层自带缓冲（flip buffer + 行规程），硬件字节先落在内核里，`read()` 只是取数，不会像 MCU 寄存器那样被覆盖；`uart_read` 非阻塞（VMIN=0 / VTIME=1，最多等 0.1s），无数据返回 0；
- **STM32 / MCU：必须软件缓冲**。硬件只有约 1 字节的 DR 寄存器，ISR 不立即取走就会被下一个字节覆盖，所以用中断 + 256 字节环形缓冲（`UART_RX_BUF_SIZE`）：
  - 中断入口：CubeMX 全局中断自动走 `HAL_UART_RxCpltCallback`（已实现）；其他芯片 / 裸机在 ISR 里调 `uart_mcu_rx_byte(&hw, byte)` 即可，不用改源码；
  - 最多 `UART_MCU_MAX`（=4）个实例，按 hw 句柄区分（&huart1、&huart2… 各自独立）；
  - 环形缓冲满时丢新字节（rx_push 有尾部保护）。

> 补充：Linux 若长期高速灌数而应用来不及读，内核缓冲耗尽同样会丢字节；此时可在应用层加"读线程 + 环形缓冲"作保险，属可选设计，非驱动必需。

## 7. 命名规范

| 前缀 | 含义 | 例子 |
|---|---|---|
| uart_ | 公开跨平台 API / 类型 | uart_open、uart_t、uart_ops_t |
| uart_mcu_ / uart_linux_ | 后端公开符号 | uart_mcu_rx_byte（中断入口） |
| dev_ | 设备管理公开 API（注册 / 查找 / 注销） | dev_register、dev_find、dev_open |
| devc_ | device.c 文件内私有静态函数 | devc_slot、devc_empty |
| mcu_ / linux_ | 后端文件内私有静态函数 | mcu_rx_push、linux_baud_to_speed |
| UART_MCU_* | MCU 配置宏（全部在 uart.h「② 硬件适配配置区」，换硬件只改这里） | UART_MCU_HANDLE_T、UART_MCU_CLK_ENABLE |

## 8. 注意事项

- 本 runner 环境没有厂商 HAL 头文件：点「检查」时 uart_mcu.c 会报 `stm32f1xx_hal.h not found`，属预期；拷进 STM32 工程即可编译；换芯片改 uart.h「② 硬件适配配置区」的 `UART_MCU_HAL_HEADER`（如 `stm32f4xx_hal.h`）；
- `uart_linux.c` 的 B460800 及以上波特率常量为 Linux 专有，已用 `#ifdef __linux__` 保护，macOS 编译不受影响；
- Linux 设备节点：`/dev/ttySx`（片上 UART）、`/dev/ttyUSBx`（USB 转串）、`/dev/ttyAMAx`、`/dev/ttyTHSx`；
- 设备节点 / `ptsname()` 等返回**静态缓冲**的字符串，传给 `uart_open` / `dev_register` 前须自行拷贝（第二次调用会覆盖第一次的结果）；
- 默认 8N1；如需数据位/校验/停止位，实现预留的 `set_config` 即可。
## 9. 开发规范（给维护者）

### 9.1 代码块（文件）分割原则

- 本驱动在代码库里按「**一文件 = 一个代码块（Fragment）**」组织：uart.h / uart.c / device.h / device.c / uart_linux.c / uart_mcu.c / README.md 各占一段；新增文件就新增一个 Fragment，不要混进已有文件；
- 归属判断：**跨平台通用** → uart.h / uart.c；**平台相关** → uart_<平台>.c；**应用代码** → 留在使用方工程；
- 新增后端（NuttX / RT-Thread / 裸机寄存器版…）：新建 uart_<平台>.c，实现 `uart_ops_t` 的 4 个函数并导出一份 `uart_default_ops`，uart.h / uart.c 零改动。

### 9.2 新增波特率（两步）

1. uart.h「① 通用配置」目录加一行：`#define UART_BAUD_<数值> <数值>`；
2. uart_linux.c `baud_map[]` 表加一行：`{ UART_BAUD_<数值>, B<数值> }`（无对应 termios 常量的平台不写，或放入 `#ifdef __linux__` 段）。

### 9.3 命名规范

- 前缀规则见第 7 章表格；补充：除 uart.h 声明的公开符号外，**一切函数/变量一律 static**，文件内私有函数用 `mcu_` / `linux_` / `devc_` 前缀，杜绝符号泄漏到链接器。

### 9.4 代码风格

- 大括号：Allman（块左大括号独占一行）；控制语句（if/for/while）**恒加花括号**，单语句也不例外；
- 缩进 2 空格，行宽约 100 列；
- 波特率用 `uart_baud_t`、收发长度用 `uart_len_t`，不裸用 uint32_t / uint16_t。

### 9.5 注释规范

- 文件头：`@file` / `@brief` / 职责 / 数据流或移植说明；
- 函数：至少 `@brief`，有入参加 `@param`，有返回值加 `@return`，坑 / 边界 / 平台差异加 `@note`；
- 配置宏：写清"改这里会怎样"（换芯片 / 换引脚 / 换厂商）的示例。

### 9.6 接口契约（冻结）

- `uart_open / uart_write / uart_read / uart_close` 4 个 API 的签名与语义**只增不改**，应用层依赖它"写一次到处跑"；
- 新能力（数据位/校验、格式化输出、缓冲查询等）走 `uart_ops_t` 预留槽位：先加签名、再加实现。

## 10. 设备管理（注册 / 查找 / 然后开发）

在驱动之上加一层"具名设备表"：驱动就绪后注册进设备管理，业务代码只认名字，不关心后端与硬件句柄。

### 10.1 接口输入 / 输出

| 接口 | 输入（参数） | 输出（返回值） | 说明 |
|---|---|---|---|
| dev_register | name=设备名, ops=后端表(NULL=默认), hw=平台句柄 | **dev_entry_t\*** 表项；失败 **NULL**（重名/表满/参数非法） | 登记设备（不打开） |
| dev_find | name=设备名 | **dev_entry_t\***；未注册 **NULL** | 按名取设备 |
| dev_open | d=表项, baud=波特率 | 同 uart_open：**0**/**-1** | 打开（绑定该设备自己的 ops） |
| dev_write / dev_read | 同 uart_write / uart_read | 同 uart_* | 便捷收发（作用于 d->u） |
| dev_close | d=表项 | 无 | 关闭设备 |
| dev_unregister | name=设备名 | 无 | 注销（未关先关，之后同名可重注册） |
| dev_count / dev_name | - / i=序号 | 数量 / 名字或 NULL | 枚举设备列表 / 调试界面 |
| uart_open_ops | u, ops, hw, baud | 同 uart_open | 指定后端打开（设备管理靠它实现"每台设备一个后端"） |

> 配置：device.h「① 配置区」—— DEV_MAX=8 个设备、DEV_NAME_MAX=16 字节名字；静态数组、无动态内存。

### 10.2 使用示例（注册 → find → 开发）

```c
#include "device.h"   /* 内含 uart.h */

int main(void)
{
    /* 1) 上电初始化：把两路串口注册成具名设备 */
    dev_register("uart-gps", NULL, &huart1);        /* STM32：HAL 句柄 */
    dev_register("uart-dbg", NULL, "/dev/ttyUSB0"); /* Linux：设备节点 */

    /* 2) 业务代码：只认名字，不关心后端 / 硬件 */
    dev_entry_t *gps = dev_find("uart-gps");
    if (gps && dev_open(gps, UART_DEFAULT_BAUD) == 0)
    {
        dev_write(gps, "AT+CGMI\r\n", 9);
        char line[64];
        int n = dev_read(gps, line, sizeof(line));
    }

    /* 3) 用完注销（未打开的设备也会安全回收） */
    dev_unregister("uart-dbg");
    return 0;
}
```

### 10.3 说明

- 注册**不打开**（打开由 dev_open 显式完成），便于"先登记、后按需使用"；
- 容量 / 名字长度都在 device.h ① 配置区，纯静态表，任何平台可编译；
- MCU 侧同时打开的实例数受 `UART_MCU_MAX` 约束（每台设备占用一个实例槽）；
- 同名重复注册被拒绝；注销后同名可重新注册；`dev_find` 未命中返回 NULL。

```
