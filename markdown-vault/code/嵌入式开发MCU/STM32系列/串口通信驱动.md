## Fragment: uart.h
```c_cpp
#ifndef UART_H
#define UART_H

#include <stdint.h>
#include <stddef.h>

/* ===================================================================
 * 通用串口(UART)驱动 —— MCU + 嵌入式 Linux 高复用（简洁版）
 *
 * 只保留 4 个核心接口：open / read / write / close。
 * 其余能力已注释预留（见 uart_ops_t 内注释），后续按需开发。
 *
 * 分层设计：应用层只调 uart_xxx()；平台差异封在后端函数表
 * uart_ops_t，换平台 = 换一个后端 .c，应用代码一行不改。
 *
 * 后端（任选其一编译，都导出 uart_default_ops）：
 *   uart_linux.c  → 嵌入式 Linux 用户态（POSIX termios，/dev/ttySx 等）
 *   uart_mcu.c    → MCU：-DUSE_STM32_HAL 用 STM32 HAL
 *
 * 用法（所有平台完全一致）：
 *   uart_t u;
 *   uart_open (&u, "/dev/ttyUSB0", 115200);   // Linux
 *   uart_open (&u, &huart1,        115200);   // STM32
 *   uart_write(&u, "AT+CGMI\r\n", 11);
 *   int n = uart_read (&u, buf, sizeof(buf));
 *   uart_close(&u);
 * =================================================================== */

typedef struct uart uart_t;

/* ---- 平台底层操作表（各后端实现） ---- */
typedef struct {
    int  (*open) (uart_t *u, const void *hw, uint32_t baud); /* 0=成功，-1=失败 */
    int  (*write)(uart_t *u, const void *buf, uint16_t len);  /* 实际发送字节数 / -1 */
    int  (*read) (uart_t *u, void *buf, uint16_t len);        /* 实际接收字节数 / -1 */
    void (*close)(uart_t *u);

    /* ===== 预留接口（后续开发时再加函数指针） =====
     * int  (*set_config)(uart_t*, uint8_t data_bits, char parity,
     *                    uint8_t stop_bits, uint8_t blocking);   // 数据位/校验/停止位/阻塞
     * int  (*printf)(uart_t*, const char *fmt, ...);             // 格式化发送
     * int  (*avail)(uart_t*);                                    // 接收缓冲可读字节数
     * int  (*flush)(uart_t*);                                    // 清空接收缓冲
     */
} uart_ops_t;

struct uart {
    const uart_ops_t *ops;   /* 后端函数表 */
    void *hw;                /* 句柄：Linux=文件描述符 / STM32=UART_HandleTypeDef* */
};

extern const uart_ops_t uart_default_ops;  /* 在所选后端 .c 中定义 */

/* ---- 应用层统一接口（仅此 4 个） ---- */
int  uart_open (uart_t *u, const void *hw, uint32_t baud);
int  uart_write(uart_t *u, const void *buf, uint16_t len);
int  uart_read (uart_t *u, void *buf, uint16_t len);
void uart_close(uart_t *u);

#endif /* UART_H */
```

## Fragment: uart.c
```c_cpp
/*
 * uart.c —— 串口驱动 · 通用核心（跨平台包装层）
 *
 * 应用层 uart_xxx() 的公共实现：绑定后端的 uart_default_ops，
 * 再转发到底层函数表。此文件与平台无关，任何平台原样编译。
 */
#include "uart.h"

#include <string.h>

int uart_open(uart_t *u, const void *hw, uint32_t baud) {
  if (!u || !hw)
    return -1;
  memset(u, 0, sizeof(*u));
  u->ops = &uart_default_ops; /* 绑定所选后端 */
  return u->ops->open(u, hw, baud);
}

int uart_write(uart_t *u, const void *buf, uint16_t len) {
  return (u && u->ops && u->ops->write) ? u->ops->write(u, buf, len) : -1;
}

int uart_read(uart_t *u, void *buf, uint16_t len) {
  return (u && u->ops && u->ops->read) ? u->ops->read(u, buf, len) : -1;
}

void uart_close(uart_t *u) {
  if (u && u->ops && u->ops->close)
    u->ops->close(u);
}
```

## Fragment: uart_linux.c
```c_cpp
/*
 * uart_linux.c —— 串口驱动 · 嵌入式 Linux 后端（POSIX termios）
 *
 * 实现 uart_default_ops；应用层代码见 uart.h，无需改动。
 * 默认配置：8N1、非阻塞读（可接收期间随时查，无数据返回 0）。
 * 设备节点：/dev/ttySx、/dev/ttyUSBx、/dev/ttyAMAx、/dev/ttyTHSx 等。
 * 依赖：仅标准 C + POSIX（open/termios/read/write）。
 */
#include "uart.h"

#if defined(__linux__)
#define _GNU_SOURCE            /* Linux 需完整 termios 特性 */
#elif defined(__APPLE__)
#define _DARWIN_C_SOURCE       /* macOS：暴露 B57600/CRTSCTS 等 BSD 定义 */
#endif
#include <errno.h>
#include <fcntl.h>
#include <termios.h>
#include <unistd.h>

/* 波特率 -> termios 常量（B460800 及以上为 Linux 专有，macOS 无定义） */
static speed_t baud_to_speed(uint32_t baud)
{
    switch (baud) {
    case 0:       return B0;
    case 50:      return B50;
    case 75:      return B75;
    case 110:     return B110;
    case 134:     return B134;
    case 150:     return B150;
    case 200:     return B200;
    case 300:     return B300;
    case 600:     return B600;
    case 1200:    return B1200;
    case 1800:    return B1800;
    case 2400:    return B2400;
    case 4800:    return B4800;
    case 9600:    return B9600;
    case 19200:   return B19200;
    case 38400:   return B38400;
    case 57600:   return B57600;
    case 115200:  return B115200;
    case 230400:  return B230400;
#ifdef __linux__
    case 460800:  return B460800;
    case 500000:  return B500000;
    case 576000:  return B576000;
    case 921600:  return B921600;
    case 1000000: return B1000000;
    case 1152000: return B1152000;
    case 1500000: return B1500000;
    case 2000000: return B2000000;
    case 2500000: return B2500000;
    case 3000000: return B3000000;
    case 3500000: return B3500000;
    case 4000000: return B4000000;
#endif
    default:      errno = EINVAL; return B0;
    }
}

static int linux_open(uart_t *u, const void *hw, uint32_t baud)
{
    const char *dev = (const char *)hw;
    if (!dev) { errno = EINVAL; return -1; }

    /* O_NOCTTY：不把设备当控制终端；O_NDELAY：打开时不等待 DCD */
    int fd = open(dev, O_RDWR | O_NOCTTY | O_NDELAY);
    if (fd < 0) return -1;

    /* 恢复为阻塞模式，实际读行为由 termios 的 VMIN/VTIME 决定 */
    if (fcntl(fd, F_SETFL, 0) < 0) { close(fd); return -1; }

    struct termios t;
    if (tcgetattr(fd, &t) != 0) { close(fd); return -1; }
    cfmakeraw(&t); /* 原始模式：关回显/ICANON/信号/软流控 */

    /* 固定 8N1、关硬件流控 */
    t.c_cflag &= ~(CSIZE | PARENB | PARODD | CSTOPB | CRTSCTS);
    t.c_cflag |= CS8;

    speed_t spd = baud_to_speed(baud);
    if (spd == B0) { close(fd); return -1; }
    if (cfsetispeed(&t, spd) || cfsetospeed(&t, spd)) { close(fd); return -1; }

    /* 非阻塞读：最多等 0.1s，无数据返回 0 */
    t.c_cc[VMIN] = 0;
    t.c_cc[VTIME] = 1;

    if (tcsetattr(fd, TCSANOW, &t) != 0) { close(fd); return -1; }
    tcflush(fd, TCIOFLUSH);
    u->hw = (void *)(intptr_t)fd;   /* 打开后 hw 保存文件描述符 */
    return 0;
}

static int linux_write(uart_t *u, const void *buf, uint16_t len)
{
    return (int)write((int)(intptr_t)u->hw, buf, len);
}

static int linux_read(uart_t *u, void *buf, uint16_t len)
{
    return (int)read((int)(intptr_t)u->hw, buf, len);
}

static void linux_close(uart_t *u)
{
    if (u->hw) { close((int)(intptr_t)u->hw); u->hw = NULL; }
}

const uart_ops_t uart_default_ops = {
    .open  = linux_open,
    .write = linux_write,
    .read  = linux_read,
    .close = linux_close,
};
```

## Fragment: uart_mcu.c
```c_cpp
/*
 * uart_mcu.c —— 串口驱动 · MCU 后端（HAL 库，跨芯片可移植版）
 *
 * ★ 移植原则：换芯片 / 换开发板，只改下面「① 移植配置区」里的宏，② 的实现一行不用动 ★
 *
 *   ① UART_MCU_HAL_HEADER        芯片 HAL 头文件（F1 默认；F4/G0/L4… 换文件名即可）
 *   ② UART_MCU_HANDLE_T          外设句柄类型（换芯片类型时只改这一行宏）
 *   ③ UART_MCU_TX/RX_PORT·PIN    串口引脚（换板子只改引脚宏）
 *   ④ UART_MCU_CLK_ENABLE()      外设时钟使能（换串口/时钟线只改这一行）
 *      UART_MCU_GPIO_CLK_ENABLE() GPIO 时钟使能（换端口如 GPIOB 时同步改）
 *   ⑤ UART_MCU_AF                复用功能号（F1 无 AF 保持默认 0；F4/H7 填如 GPIO_AF7_USART1）
 *
 * 使用（应用层与平台无关，见 uart.h）：
 *   uart_open(&u, &huart1, 115200);
 * 中断接收：
 *   - STM32 + CubeMX 全局中断：HAL_UART_RxCpltCallback 已自动接管，无需额外代码；
 *   - 其他芯片 / 裸机中断：在 ISR 里调 uart_mcu_rx_byte(&huart1, byte) 即可，与芯片无关。
 */
#include "uart.h"

/* ==================== ① 移植配置区 ★换硬件只改这里★ ==================== */

#ifndef UART_MCU_HAL_HEADER
#define UART_MCU_HAL_HEADER "stm32f1xx_hal.h"   /* 例: "stm32f4xx_hal.h" / "stm32g0xx_hal.h" */
#endif
#include UART_MCU_HAL_HEADER

/* 外设句柄类型：换芯片类型只改这一行（GD32、NXP 等换成自己的句柄类型） */
#ifndef UART_MCU_HANDLE_T
#define UART_MCU_HANDLE_T UART_HandleTypeDef
#endif

/* 串口引脚配置：默认 USART1 = PA9(TX) / PA10(RX)，换板子只改这里 */
#ifndef UART_MCU_TX_PORT
#define UART_MCU_TX_PORT GPIOA
#define UART_MCU_TX_PIN  GPIO_PIN_9
#define UART_MCU_RX_PORT GPIOA
#define UART_MCU_RX_PIN  GPIO_PIN_10
#endif

/* 时钟使能：换串口外设 / 时钟线只改这两行
 * 例（GD32）: #define UART_MCU_CLK_ENABLE()      rcu_periph_clock_enable(RCU_USART1)
 *             #define UART_MCU_GPIO_CLK_ENABLE() rcu_periph_clock_enable(RCU_GPIOA) */
#ifndef UART_MCU_CLK_ENABLE
#define UART_MCU_CLK_ENABLE()      __HAL_RCC_USART1_CLK_ENABLE()
#endif
#ifndef UART_MCU_GPIO_CLK_ENABLE
#define UART_MCU_GPIO_CLK_ENABLE() __HAL_RCC_GPIOA_CLK_ENABLE()
#endif

/* 复用功能号：F1 无 AF 概念保持默认 0（自动跳过 Alternate 域）；F4/H7 等填如 GPIO_AF7_USART1 */
#ifndef UART_MCU_AF
#define UART_MCU_AF 0
#endif

/* ==================== ② 通用实现区（移植无需改动） ==================== */

#define UART_MCU_MAX 4
#define UART_RX_BUF_SIZE 256

/* ------------------------- 环形缓冲（中断接收用，与芯片无关） ------------------------- */
typedef struct {
  UART_MCU_HANDLE_T *huart; /* 绑定的外设句柄 */
  uint8_t rx[UART_RX_BUF_SIZE];
  volatile uint16_t head, tail;
  uint8_t used;
} uart_inst_t;

static uart_inst_t insts[UART_MCU_MAX];

static uart_inst_t *inst_find(const void *hw) {
  for (int i = 0; i < UART_MCU_MAX; i++)
    if (insts[i].used && insts[i].huart == hw)
      return &insts[i];
  return NULL;
}
static uart_inst_t *inst_alloc(void) {
  for (int i = 0; i < UART_MCU_MAX; i++)
    if (!insts[i].used) {
      insts[i].used = 1;
      insts[i].head = insts[i].tail = 0;
      return &insts[i];
    }
  return NULL;
}
static void rx_push(uart_inst_t *i, uint8_t b) {
  uint16_t nxt = (i->head + 1) & (UART_RX_BUF_SIZE - 1);
  if (nxt != i->tail) {
    i->rx[i->head] = b;
    i->head = nxt;
  } /* 满则丢 */
}
static int rx_pop(uart_inst_t *i, uint8_t *b) {
  if (i->head == i->tail)
    return 0;
  *b = i->rx[i->tail];
  i->tail = (i->tail + 1) & (UART_RX_BUF_SIZE - 1);
  return 1;
}

/* ------------------------- 硬件初始化：全部由配置区宏驱动 ------------------------- */
static void uart_mcu_hw_init(void) {
  UART_MCU_CLK_ENABLE();      /* 外设时钟：换芯片/时钟线只改宏 */
  UART_MCU_GPIO_CLK_ENABLE(); /* GPIO 时钟：换到 GPIOB 等端口时同步改宏 */

  GPIO_InitTypeDef g;
  g.Pin = UART_MCU_TX_PIN;
  g.Mode = GPIO_MODE_AF_PP;
  g.Pull = GPIO_NOPULL;
  g.Speed = GPIO_SPEED_FREQ_HIGH;
#if UART_MCU_AF
  g.Alternate = UART_MCU_AF;  /* F4/H7 等有复用号；F1 无该成员，AF=0 自动跳过 */
#endif
  HAL_GPIO_Init(UART_MCU_TX_PORT, &g);

  g.Pin = UART_MCU_RX_PIN;
  HAL_GPIO_Init(UART_MCU_RX_PORT, &g);
}

/* ------------------------- 后端 4 接口 ------------------------- */
static int mcu_open(uart_t *u, const void *hw, uint32_t baud) {
  UART_MCU_HANDLE_T *huart = (UART_MCU_HANDLE_T *)hw;
  if (!huart)
    return -1;
  uart_inst_t *i = inst_alloc();
  if (!i)
    return -1;
  i->huart = huart;
  u->hw = (void *)hw;

  uart_mcu_hw_init(); /* 开时钟 + 配 GPIO 复用引脚（宏驱动，幂等） */

  huart->Init.BaudRate = baud; /* 固定 8N1 */
  huart->Init.WordLength = UART_WORDLENGTH_8B;
  huart->Init.Parity = UART_PARITY_NONE;
  huart->Init.StopBits = UART_STOPBITS_1;
  huart->Init.Mode = UART_MODE_TX_RX;
  huart->Init.HwFlowCtl = UART_HWCONTROL_NONE;
  huart->Init.OverSampling = UART_OVERSAMPLING_16;
  if (HAL_UART_Init(huart) != HAL_OK) {
    i->used = 0;
    return -1;
  }

  /* 启动"每收 1 字节进环形缓冲"的中断接收 */
  HAL_UART_Receive_IT(huart, (uint8_t *)&huart->Instance->DR, 1);
  return 0;
}

static int mcu_write(uart_t *u, const void *buf, uint16_t len) {
  return HAL_UART_Transmit((UART_MCU_HANDLE_T *)u->hw, (const uint8_t *)buf, len, 1000) == HAL_OK
             ? (int)len
             : -1;
}

static int mcu_read(uart_t *u, void *buf, uint16_t len) {
  uart_inst_t *i = inst_find(u->hw);
  if (!i)
    return -1;
  uint8_t *p = (uint8_t *)buf;
  uint16_t n = 0;
  while (n < len) {
    uint8_t b;
    if (!rx_pop(i, &b))
      break; /* 非阻塞：读空即返回 */
    p[n++] = b;
  }
  return n;
}

static void mcu_close(uart_t *u) {
  uart_inst_t *i = inst_find(u->hw);
  if (i) {
    HAL_UART_AbortReceive((UART_MCU_HANDLE_T *)u->hw);
    i->used = 0;
  }
  u->hw = NULL;
}

/* ------------------------- 中断接收入口（与芯片无关） ------------------------- */

/* 任何芯片的 ISR 都可以直接调用：收到 1 字节 → 入对应实例的环形缓冲 */
void uart_mcu_rx_byte(const void *hw, uint8_t b) {
  uart_inst_t *i = inst_find(hw);
  if (i)
    rx_push(i, b);
}

/* STM32 HAL 专用：CubeMX 全局中断自动走到这里（F1/F4/L4… 通用） */
void HAL_UART_RxCpltCallback(UART_HandleTypeDef *huart) {
  uart_mcu_rx_byte((const void *)huart, (uint8_t)(huart->Instance->DR & 0xFF));
  HAL_UART_Receive_IT(huart, (uint8_t *)&huart->Instance->DR, 1); /* 继续收 */
}

const uart_ops_t uart_default_ops = {
    .open = mcu_open,
    .write = mcu_write,
    .read = mcu_read,
    .close = mcu_close,
};
```

## Fragment: README.md
```markdown
# 通用串口(UART)驱动 —— MCU + 嵌入式 Linux 高复用（简洁版）

一套接口，两种平台（嵌入式 Linux / STM32 MCU）。应用层写一次，换平台只换后端；平台差异封在函数表 `uart_ops_t` 里，各后端导出 `uart_default_ops`。

## 分层设计

| 层 | 文件 | 职责 |
|---|---|---|
| 应用层 | 你的 main.c | 只调 `uart_open/write/read/close`，不关心平台 |
| 通用核心 | uart.c | 包装函数，把调用转发到 `u->ops` 函数表 |
| 后端 | uart_linux.c / uart_mcu.c | 实现 `uart_default_ops`；二选一参与编译 |

## 接口（只保留 4 个，其余注释预留）
```c
int  uart_open (uart_t *u, const void *hw, uint32_t baud);  /* 0=成功，-1=失败 */
int  uart_write(uart_t *u, const void *buf, uint16_t len);   /* 实际发送字节数 / -1 */
int  uart_read (uart_t *u, void *buf, uint16_t len);         /* 实际接收字节数 / -1 */
void uart_close(uart_t *u);
```

- `hw`：Linux 传设备节点字符串（如 `"/dev/ttyUSB0"`）；STM32 传 HAL 句柄（如 `&huart1`）
- 默认 8N1、非阻塞读（无数据返回 0）
- 其余能力（数据位/校验/停止位 `set_config`、格式化发送 `printf`、可读字节数 `avail`、清缓冲 `flush`）已注释在 `uart_ops_t` 内，后续按需加函数指针即可

## 文件清单

| 文件 | 作用 | 平台相关? |
|------|------|:--:|
| uart.h | 统一 API + 结构体（uart_t / uart_ops_t） | 否 |
| uart.c | 通用核心：包装函数（转发到后端） | 否 |
| uart_linux.c | Linux 后端：termios，支持 /dev/ttySx、ttyUSBx、ttyAMAx、ttyTHSx | Linux |
| uart_mcu.c | MCU 后端：HAL 库（-DUSE_STM32_HAL）；顶部★移植配置区★只改宏即可换芯片/引脚/时钟，环形缓冲 + 4 接口通用 | MCU |

## 快速开始

### Linux（直接编译运行）

```bash
gcc uart.c uart_linux.c main.c -o app
./app
```

```c
#include "uart.h"
uart_t u;
uart_open (&u, "/dev/ttyUSB0", 115200);
uart_write(&u, "AT+CGMI\r\n", 11);
int n = uart_read (&u, buf, sizeof(buf));
uart_close(&u);
```

### STM32（CubeMX / CubeIDE）

工程里加 `uart.c + uart_mcu.c`，预处理器加 `-DUSE_STM32_HAL`；CubeMX 里开 USART1 + 全局中断(NVIC) 即可（**GPIO 不用手动配置**，见下一节）。应用代码与 Linux 完全相同：

```c
#include "uart.h"
uart_t u;
uart_open (&u, &huart1, 115200);   /* huart1 来自 CubeMX */
uart_write(&u, "AT+CGMI\r\n", 11);
uart_close(&u);
```

## 硬件适配（移植只需改宏）

`uart_mcu_hw_init()` 自动完成：开外设时钟 + 开 GPIO 时钟 + TX/RX 引脚配为复用（幂等，CubeMX 已配也安全；裸机也自包含）。**换芯片 / 换板子只改 `uart_mcu.c` 顶部「① 移植配置区」里的宏，下面实现一行不用动**：

| 宏 | 默认值 | 说明 |
|---|---|---|
| UART_MCU_HAL_HEADER | "stm32f1xx_hal.h" | 芯片 HAL 头文件（F4 改 "stm32f4xx_hal.h"） |
| UART_MCU_HANDLE_T | UART_HandleTypeDef | 外设句柄类型（换芯片类型只改这一行） |
| UART_MCU_TX/RX_PORT·PIN | GPIOA / GPIO_PIN_9 / GPIO_PIN_10 | TX/RX 引脚（换板子只改这里） |
| UART_MCU_CLK_ENABLE() | __HAL_RCC_USART1_CLK_ENABLE() | 外设时钟使能（换串口/时钟线只改这里） |
| UART_MCU_GPIO_CLK_ENABLE() | __HAL_RCC_GPIOA_CLK_ENABLE() | GPIO 时钟使能（换端口时同步改） |
| UART_MCU_AF | 0 | 复用功能号（F1 无 AF 保持 0；F4/H7 填 GPIO_AF7_USART1） |

移植三步：
- **换系列**（如 F1 → F4）：① `UART_MCU_HAL_HEADER` 改 "stm32f4xx_hal.h"；② `UART_MCU_AF` 填 GPIO_AF7_USART1；
- **换板子**（如 PA9/PA10 → PB6/PB7）：改 TX/RX 引脚宏 + `UART_MCU_GPIO_CLK_ENABLE()` 换成 GPIOB；
- **换芯片厂商**（如 GD32）：改 HAL 头文件 + `UART_MCU_HANDLE_T` 类型 + 两个时钟宏，中断里调 `uart_mcu_rx_byte(&hw, byte)` 收字节即可，源码不改。

## 接收机制

- 中断 + 256 字节环形缓冲（`UART_RX_BUF_SIZE`），不丢字节、不阻塞主循环；
- 中断入口：STM32 + CubeMX 全局中断自动走 `HAL_UART_RxCpltCallback`（已实现）；其他芯片 / 裸机在 ISR 里调 `uart_mcu_rx_byte(&hw, byte)` 即可，不用改源码；
- `uart_read` 非阻塞读缓冲，无数据返回 0；
- 最多 `UART_MCU_MAX`（=4）个实例，按 hw 句柄区分（&huart1、&huart2… 各自独立）；
- 环形缓冲满时丢新字节（rx_push 有尾部保护）。

## 注意事项

- 本 runner 环境没有厂商 HAL 头文件：点「检查」时 uart_mcu.c 会报 `stm32f1xx_hal.h not found`，属预期；拷进 STM32 工程即可编译；换芯片改 `UART_MCU_HAL_HEADER`（如 `stm32f4xx_hal.h`）；
- `uart_linux.c` 的 B460800 及以上波特率常量为 Linux 专有，已用 `#ifdef __linux__` 保护，macOS 编译不受影响；
- Linux 读为非阻塞（有数据立即可读）；设备节点：`/dev/ttySx`（片上 UART）、`/dev/ttyUSBx`（USB 转串）、`/dev/ttyAMAx`、`/dev/ttyTHSx`；
- 默认 8N1；如需数据位/校验/停止位，实现预留的 `set_config` 即可。
