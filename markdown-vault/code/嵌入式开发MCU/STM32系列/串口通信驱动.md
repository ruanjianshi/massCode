```c_cpp
#ifndef UART_H
#define UART_H

/**
 * @file    uart.h
 * @brief   最小 UART 统一接口
 *
 * 用同一套 open/write/read/close 接口，屏蔽 Linux termios
 * 与 STM32 HAL 两种后端的差异。
 *
 * 依赖（来自“通用接口管理”片段）：
 *   - common.h  通用错误码 COMMON_*
 *   - device.h  通用设备注册表（可选具名管理）
 */

#include <stddef.h>
#include <stdint.h>
#include "common.h"
#include "device.h"

/* =====================================================================
 * ① 公共类型
 * ===================================================================== */
typedef uint32_t uart_baud_t;   /* 波特率，单位 Hz */
typedef size_t uart_len_t;      /* 读写长度，单位字节 */

/* =====================================================================
 * ② 波特率常量
 *
 * 只列常用档位。MCU 后端直接写入 HAL 的 BaudRate 字段；
 * Linux 后端经 baud_table 映射到 termios 的 speed_t。
 * ===================================================================== */
#define UART_BAUD_9600    9600U
#define UART_BAUD_19200   19200U
#define UART_BAUD_38400   38400U
#define UART_BAUD_57600   57600U
#define UART_BAUD_115200  115200U
#define UART_BAUD_230400  230400U
#define UART_BAUD_460800  460800U
#define UART_BAUD_921600  921600U

/* =====================================================================
 * ③ 发送超时（仅 MCU 后端阻塞发送使用）
 * ===================================================================== */
#ifndef UART_MCU_TX_TIMEOUT_MS
#define UART_MCU_TX_TIMEOUT_MS 1000U
#endif

/* =====================================================================
 * ④ 设备对象（不透明句柄，结构体见 ⑥）
 * ===================================================================== */
typedef struct uart uart_t;

/* =====================================================================
 * ⑤ 后端操作表（虚函数表）
 *
 * 每个后端（uart_linux.c / uart_mcu.c）实现一份，
 * 最终二选一参与链接，默认表 uart_default_ops 指向参与链接的那个。
 * ===================================================================== */
typedef struct {
    int (*open)(uart_t *uart, const void *hw, uart_baud_t baud);
    int (*write)(uart_t *uart, const void *buf, uart_len_t len);
    int (*read)(uart_t *uart, void *buf, uart_len_t len);
    int (*close)(uart_t *uart);
} uart_ops_t;

/* =====================================================================
 * ⑥ 设备对象结构
 *
 * backend：后端私有句柄，编码方式由各后端自定
 *          （Linux 用 fd+1 避开 0 号句柄；MCU 用实例数组指针）。
 * opened ：1=已打开，重复打开被拒绝（COMMON_ESTATE）。
 * ===================================================================== */
struct uart {
    const uart_ops_t *ops;
    uintptr_t backend;
    unsigned char opened;
};

/* 静态初始化器：所有 uart_t 声明都应使用它初始化 */
#define UART_INITIALIZER { NULL, 0U, 0U }

/* 默认后端操作表（由参与链接的那个后端导出） */
extern const uart_ops_t uart_default_ops;

/* =====================================================================
 * ⑦ 基础接口（统一入口，公共层负责参数检查后转发给后端）
 * ===================================================================== */

/**
 * @brief  把对象清零为未初始化状态
 * @param  uart 设备对象
 */
void uart_init(uart_t *uart);

/**
 * @brief  打开串口（使用默认后端）
 * @param  uart 已初始化的设备对象
 * @param  hw   后端硬件标识：Linux 传设备路径字符串，MCU 传 HAL 句柄指针
 * @param  baud 波特率（用 UART_BAUD_* 常量）
 * @return COMMON_OK 成功；COMMON_EINVAL 参数非法；COMMON_ESTATE 重复打开；
 *         其余为后端错误（EBUSY/ETIMEOUT/EIO…）
 * @note   打开失败会回滚对象状态，可重试。
 */
int uart_open(uart_t *uart, const void *hw, uart_baud_t baud);

/**
 * @brief  打开串口（指定操作表，用于接入自定义后端）
 * @param  uart 已初始化的设备对象
 * @param  ops  操作表（open/write/read/close 四者必须非空）
 * @param  hw   后端硬件标识
 * @param  baud 波特率
 * @return 同 uart_open
 */
int uart_open_ops(uart_t *uart, const uart_ops_t *ops,
                  const void *hw, uart_baud_t baud);

/**
 * @brief  发送数据
 * @param  uart 已打开的设备对象
 * @param  buf  数据缓冲区（len 为 0 时可为 NULL）
 * @param  len  期望发送字节数
 * @return 成功返回实际发送字节数（可能小于 len，上层应循环发送）；
 *         COMMON_ESTATE 未打开；COMMON_EINVAL 参数非法
 */
int uart_write(uart_t *uart, const void *buf, uart_len_t len);

/**
 * @brief  接收数据（非阻塞）
 * @param  uart 已打开的设备对象
 * @param  buf  接收缓冲区（len 为 0 时可为 NULL）
 * @param  len  缓冲区容量
 * @return 成功返回实际读到的字节数（可小于 len，0 表示暂无数据）；
 *         COMMON_ESTATE 未打开；COMMON_EINVAL 参数非法
 */
int uart_read(uart_t *uart, void *buf, uart_len_t len);

/**
 * @brief  关闭串口
 * @param  uart 设备对象
 * @return COMMON_OK；后端错误（COMMON_EIO 等）
 * @note   关闭后对象回到未打开状态，可再次 uart_open。
 */
int uart_close(uart_t *uart);

/* =====================================================================
 * ⑧ 可选：具名设备适配（把 UART 放进通用注册表按名管理）
 *
 * 只有当需要按名称查找多个设备时才使用这组接口；
 * 单串口程序直接使用 uart_* 即可。
 * ===================================================================== */

/**
 * @brief  以默认后端登记一个具名 UART
 * @param  name 设备名（须满足 dev_register 的命名规则）
 * @param  hw   后端硬件标识
 * @param  out  输出参数：成功时写入设备对象
 * @return COMMON_OK 或 dev_register 的错误码
 */
int uart_dev_register(const char *name, const void *hw, dev_entry_t **out);

/**
 * @brief  以指定操作表登记一个具名 UART
 * @param  name 设备名
 * @param  ops  操作表（四方法必填）
 * @param  hw   后端硬件标识
 * @param  out  输出参数：成功时写入设备对象
 * @return COMMON_OK；COMMON_EINVAL/COMMON_ENOSYS/COMMON_ENOSPC…
 */
int uart_dev_register_ops(const char *name, const uart_ops_t *ops,
                          const void *hw, dev_entry_t **out);

/** @brief 打开具名设备（用登记时保存的 ops 与 config） */
int uart_dev_open(dev_entry_t *dev, uart_baud_t baud);
/** @brief 向具名设备发送 */
int uart_dev_write(dev_entry_t *dev, const void *buf, uart_len_t len);
/** @brief 从具名设备接收 */
int uart_dev_read(dev_entry_t *dev, void *buf, uart_len_t len);
/** @brief 关闭具名设备 */
int uart_dev_close(dev_entry_t *dev);

/**
 * @brief  关闭并从注册表注销具名设备
 * @note   内部依次：关闭 UART → 注销注册表 → 归还槽位。
 */
int uart_dev_unregister(dev_entry_t *dev);

/* =====================================================================
 * ⑨ MCU 中断接收回调转发入口
 *
 * 在 HAL_UART_RxCpltCallback 中调用：把刚收到的单字节
 * 压入对应实例的环形缓冲，并重新启动下一字节接收。
 * ===================================================================== */
void uart_mcu_rx_complete(const void *hw);

/* =====================================================================
 * ⑩ 扩展预留
 * ===================================================================== */
/*
 * - 超时读写、DMA 收发
 * - 统计与自动错误恢复
 * - RTOS 锁与异步回调
 */

#endif /* UART_H */
```

```c_cpp
/**
 * @file    uart.c
 * @brief   UART 基本接口分发
 *
 * 公共层的职责：
 * 1. 参数与状态检查；
 * 2. 校验操作表完整性；
 * 3. 把调用转发给后端 ops；
 * 4. 失败时回滚对象状态。
 */

#include <limits.h>
#include <string.h>
#include "uart.h"

/* =====================================================================
 * ① 对象初始化
 * ===================================================================== */
void uart_init(uart_t *uart)
{
    if (uart != NULL)
        memset(uart, 0, sizeof(*uart));
}

/* =====================================================================
 * ② 打开（指定操作表）
 * ===================================================================== */
int uart_open_ops(uart_t *uart, const uart_ops_t *ops,
                  const void *hw, uart_baud_t baud)
{
    int result;

    /* 参数校验：对象/操作表/硬件非空，波特率非 0 */
    if (uart == NULL || ops == NULL || hw == NULL || baud == 0U)
        return COMMON_EINVAL;
    /* 操作表完整性：四个必选方法都不能缺 */
    if (ops->open == NULL || ops->write == NULL ||
        ops->read == NULL || ops->close == NULL)
        return COMMON_ENOSYS;
    /* 重复打开保护 */
    if (uart->opened)
        return COMMON_ESTATE;

    /* 先绑定操作表，再调用后端 open；
     * 后端 open 失败时把对象恢复到“未绑定”状态，保证可重试。 */
    uart->ops = ops;
    uart->backend = 0U;
    result = ops->open(uart, hw, baud);
    if (result < 0) {
        uart->ops = NULL;
        uart->backend = 0U;
        return result;
    }
    uart->opened = 1U;
    return COMMON_OK;
}

/* =====================================================================
 * ③ 打开（默认后端）
 * ===================================================================== */
int uart_open(uart_t *uart, const void *hw, uart_baud_t baud)
{
    return uart_open_ops(uart, &uart_default_ops, hw, baud);
}

/* =====================================================================
 * ④ 发送
 * ===================================================================== */
int uart_write(uart_t *uart, const void *buf, uart_len_t len)
{
    /* len 为 0 时允许 buf 为 NULL；长度上限受 int 返回值约束 */
    if (uart == NULL || (buf == NULL && len != 0U) ||
        len > (uart_len_t)INT_MAX)
        return COMMON_EINVAL;
    if (!uart->opened || uart->ops == NULL)
        return COMMON_ESTATE;
    if (len == 0U)
        return 0;                    /* 空发送直接成功 */
    return uart->ops->write(uart, buf, len);
}

/* =====================================================================
 * ⑤ 接收
 * ===================================================================== */
int uart_read(uart_t *uart, void *buf, uart_len_t len)
{
    if (uart == NULL || (buf == NULL && len != 0U) ||
        len > (uart_len_t)INT_MAX)
        return COMMON_EINVAL;
    if (!uart->opened || uart->ops == NULL)
        return COMMON_ESTATE;
    if (len == 0U)
        return 0;                    /* 空接收直接成功 */
    return uart->ops->read(uart, buf, len);
}

/* =====================================================================
 * ⑥ 关闭
 * ===================================================================== */
int uart_close(uart_t *uart)
{
    int result;

    if (uart == NULL)
        return COMMON_EINVAL;
    /* 未打开视为关闭成功（幂等） */
    if (!uart->opened)
        return COMMON_OK;

    /* 先调后端 close，再清理公共层状态；
     * 即使后端 close 报错，也把对象置回未打开，避免悬空。 */
    result = uart->ops->close(uart);
    uart->ops = NULL;
    uart->backend = 0U;
    uart->opened = 0U;
    return result;
}
```

```c_cpp
/**
 * @file    uart_dev.c
 * @brief   UART 与简单设备注册表的适配
 *
 * 为每个具名 UART 分配一个静态槽位（内嵌 uart_t），
 * 并把“槽位地址”作为 data 存入注册表，这样：
 *   dev_find(name) → dev_get_data(dev) → uart_slot_t → uart_* 调用。
 */

#include <string.h>
#include "uart.h"

/* =====================================================================
 * ① 槽位池
 * ===================================================================== */
typedef struct {
    uart_t uart;              /* 内嵌的 UART 对象 */
    unsigned char used;       /* 1=已占用 0=空闲 */
} uart_slot_t;

static uart_slot_t slots[DEV_MAX];

/* =====================================================================
 * ② 内部工具：由设备对象找回槽位
 * ===================================================================== */

/* 先校验类别是 UART，再用 data 指针反向定位槽位（线性查找） */
static uart_slot_t *slot_from_dev(dev_entry_t *dev)
{
    void *data;
    int i;

    if (dev_get_class(dev) != DEV_CLASS_UART)
        return NULL;
    data = dev_get_data(dev);
    for (i = 0; i < DEV_MAX; i++)
        if (&slots[i] == data && slots[i].used)
            return &slots[i];
    return NULL;
}

/* =====================================================================
 * ③ 注册
 * ===================================================================== */
int uart_dev_register_ops(const char *name, const uart_ops_t *ops,
                          const void *hw, dev_entry_t **out)
{
    int i;
    int result;

    /* 操作表与输出指针必填，硬件标识必填 */
    if (ops == NULL || hw == NULL || out == NULL)
        return COMMON_EINVAL;
    /* 操作表四方法完整性检查 */
    if (ops->open == NULL || ops->write == NULL ||
        ops->read == NULL || ops->close == NULL)
        return COMMON_ENOSYS;

    /* 找一个空闲槽位，初始化后交给注册表登记 */
    for (i = 0; i < DEV_MAX; i++) {
        if (!slots[i].used) {
            memset(&slots[i], 0, sizeof(slots[i]));
            slots[i].used = 1U;
            uart_init(&slots[i].uart);
            result = dev_register(name, DEV_CLASS_UART, ops, hw,
                                  &slots[i], out);
            if (result < 0)
                memset(&slots[i], 0, sizeof(slots[i])); /* 登记失败归还槽位 */
            return result;
        }
    }
    return COMMON_ENOSPC;   /* 槽位池满 */
}

int uart_dev_register(const char *name, const void *hw, dev_entry_t **out)
{
    return uart_dev_register_ops(name, &uart_default_ops, hw, out);
}

/* =====================================================================
 * ④ 打开 / 读写 / 关闭
 *
 * 打开时从注册表取出 ops 与 config：登记时存的 ops 决定后端，
 * config 即硬件标识（Linux 设备路径 / MCU HAL 句柄）。
 * ===================================================================== */
int uart_dev_open(dev_entry_t *dev, uart_baud_t baud)
{
    uart_slot_t *slot = slot_from_dev(dev);

    if (slot == NULL)
        return COMMON_EINVAL;
    return uart_open_ops(&slot->uart,
                         (const uart_ops_t *)dev_get_api(dev),
                         dev_get_config(dev), baud);
}

int uart_dev_write(dev_entry_t *dev, const void *buf, uart_len_t len)
{
    uart_slot_t *slot = slot_from_dev(dev);

    return slot == NULL ? COMMON_EINVAL
                        : uart_write(&slot->uart, buf, len);
}

int uart_dev_read(dev_entry_t *dev, void *buf, uart_len_t len)
{
    uart_slot_t *slot = slot_from_dev(dev);

    return slot == NULL ? COMMON_EINVAL
                        : uart_read(&slot->uart, buf, len);
}

int uart_dev_close(dev_entry_t *dev)
{
    uart_slot_t *slot = slot_from_dev(dev);

    return slot == NULL ? COMMON_EINVAL : uart_close(&slot->uart);
}

/* =====================================================================
 * ⑤ 注销
 *
 * 顺序：关闭 UART → 从注册表注销 → 清零槽位。
 * 任何一步失败都返回对应错误码（关闭错误优先）。
 * ===================================================================== */
int uart_dev_unregister(dev_entry_t *dev)
{
    uart_slot_t *slot = slot_from_dev(dev);
    int close_result;
    int result;

    if (slot == NULL)
        return COMMON_EINVAL;
    close_result = uart_close(&slot->uart);
    result = dev_unregister(dev);
    memset(slot, 0, sizeof(*slot));
    return close_result < 0 ? close_result : result;
}
```

```c_cpp
/**
 * @file    uart_linux.c
 * @brief   Linux/POSIX termios 后端
 *
 * 打开：以 O_NONBLOCK 打开设备路径并配置为原始模式 8N1。
 * 读写：非阻塞语义——无数据时返回 0，EAGAIN 同样按 0 处理，
 *       由上层决定轮询或等待。
 * 句柄编码：backend = fd + 1，避开 0 号句柄（0 视为“未打开”）。
 */

#if defined(__linux__)
#define _GNU_SOURCE
#elif defined(__APPLE__)
#define _DARWIN_C_SOURCE
#endif

#include <errno.h>
#include <fcntl.h>
#include <termios.h>
#include <unistd.h>
#include "uart.h"

/* =====================================================================
 * ① 波特率映射表
 *
 * 把公共波特率常量映射到 termios 的 speed_t。
 * 高速档（230400 及以上）在部分平台无对应宏，用 #ifdef 裁剪。
 * ===================================================================== */
typedef struct {
    uart_baud_t baud;     /* 公共波特率（Hz） */
    speed_t speed;        /* termios 波特率宏 */
} baud_item_t;

static const baud_item_t baud_table[] = {
    { UART_BAUD_9600, B9600 },
    { UART_BAUD_19200, B19200 },
    { UART_BAUD_38400, B38400 },
    { UART_BAUD_57600, B57600 },
    { UART_BAUD_115200, B115200 },
#ifdef B230400
    { UART_BAUD_230400, B230400 },
#endif
#ifdef B460800
    { UART_BAUD_460800, B460800 },
#endif
#ifdef B921600
    { UART_BAUD_921600, B921600 },
#endif
};

/* 查表：支持则返回 COMMON_OK 并给出 speed，否则 COMMON_EINVAL */
static int baud_speed(uart_baud_t baud, speed_t *speed)
{
    size_t i;

    for (i = 0; i < sizeof(baud_table) / sizeof(baud_table[0]); i++) {
        if (baud_table[i].baud == baud) {
            *speed = baud_table[i].speed;
            return COMMON_OK;
        }
    }
    return COMMON_EINVAL;
}

/* =====================================================================
 * ② 私有工具
 * ===================================================================== */

/* backend 存的是 fd+1，取出真实 fd；0 表示未打开返回 -1 */
static int linux_fd(const uart_t *uart)
{
    return uart->backend == 0U ? -1 : (int)(uart->backend - 1U);
}

/* =====================================================================
 * ③ 后端实现（作为 uart_default_ops 的四个方法）
 * ===================================================================== */
static int linux_open(uart_t *uart, const void *hw, uart_baud_t baud)
{
    const char *path = (const char *)hw;
    struct termios config;
    speed_t speed;
    int fd;

    /* 路径与波特率校验 */
    if (path == NULL || baud_speed(baud, &speed) < 0)
        return COMMON_EINVAL;

    /* 非阻塞 + 不占控制终端打开 */
    fd = open(path, O_RDWR | O_NOCTTY | O_NONBLOCK);
    if (fd < 0)
        return errno == EBUSY ? COMMON_EBUSY : COMMON_EIO;
    if (tcgetattr(fd, &config) != 0) {
        close(fd);
        return COMMON_EIO;
    }

    /* 原始模式 8N1：清掉流控，读写均不等待（VMIN/VTIME=0） */
    cfmakeraw(&config);
    config.c_cflag |= CLOCAL | CREAD;
    config.c_cflag &= ~CSTOPB;
#ifdef CRTSCTS
    config.c_cflag &= ~CRTSCTS;
#endif
    config.c_cc[VMIN] = 0;
    config.c_cc[VTIME] = 0;

    if (cfsetispeed(&config, speed) != 0 ||
        cfsetospeed(&config, speed) != 0 ||
        tcsetattr(fd, TCSANOW, &config) != 0) {
        close(fd);
        return COMMON_EIO;
    }

    /* 打开成功才写 backend；之前所有失败路径都已 close(fd) */
    uart->backend = (uintptr_t)(fd + 1);
    return COMMON_OK;
}

static int linux_write(uart_t *uart, const void *buf, uart_len_t len)
{
    ssize_t count;

    /* EINTR 表示被信号打断，重试一次即可 */
    do {
        count = write(linux_fd(uart), buf, len);
    } while (count < 0 && errno == EINTR);

    if (count >= 0)
        return (int)count;
    /* 非阻塞写满时按 0 字节处理，由上层稍后重发 */
    return (errno == EAGAIN || errno == EWOULDBLOCK) ? 0 : COMMON_EIO;
}

static int linux_read(uart_t *uart, void *buf, uart_len_t len)
{
    ssize_t count;

    do {
        count = read(linux_fd(uart), buf, len);
    } while (count < 0 && errno == EINTR);

    if (count >= 0)
        return (int)count;
    /* 非阻塞暂无数据按 0 处理（与底层返回一致） */
    return (errno == EAGAIN || errno == EWOULDBLOCK) ? 0 : COMMON_EIO;
}

static int linux_close(uart_t *uart)
{
    int fd = linux_fd(uart);

    /* 先清 backend 再 close，即使 close 出错对象也不留悬空句柄 */
    uart->backend = 0U;
    return fd < 0 || close(fd) == 0 ? COMMON_OK : COMMON_EIO;
}

/* =====================================================================
 * ④ 默认后端表
 * ===================================================================== */
const uart_ops_t uart_default_ops = {
    linux_open,
    linux_write,
    linux_read,
    linux_close
};
```

```c_cpp
/**
 * @file    uart_mcu.c
 * @brief   STM32 HAL 后端：阻塞发送 + 中断环形缓冲接收
 *
 * GPIO、时钟、NVIC 和 HAL 句柄由 Board/CubeMX 配置，本驱动不硬编码
 * 具体 USART 与引脚（符合“应用/驱动不碰板级资源”的分层规则）。
 *
 * 接收链路：
 *   HAL_UART_Receive_IT(单字节) → ISR 回调 uart_mcu_rx_complete
 *   → rx_push 压入环形缓冲 → 任务侧 uart_read 取出
 */

#include <string.h>
#include "uart.h"

/* =====================================================================
 * ① 适配区（编译期可裁剪，见 uart.h 顶部说明）
 * ===================================================================== */
#ifndef UART_MCU_HAL_HEADER
#define UART_MCU_HAL_HEADER "stm32f1xx_hal.h"
#endif
#include UART_MCU_HAL_HEADER

#ifndef UART_MCU_HANDLE_T
#define UART_MCU_HANDLE_T UART_HandleTypeDef
#endif

#ifndef UART_MCU_MAX
#define UART_MCU_MAX 4          /* 最多同时打开几路 UART */
#endif

#ifndef UART_RX_BUF_SIZE
#define UART_RX_BUF_SIZE 128    /* 每路接收环形缓冲大小（字节） */
#endif

/* 非法配置直接编译报错，而不是运行期才暴露 */
#if UART_MCU_MAX < 1 || UART_RX_BUF_SIZE < 2
#error "invalid UART MCU configuration"
#endif

/* =====================================================================
 * ② 私有结构
 * ===================================================================== */
typedef struct {
    UART_MCU_HANDLE_T *handle;      /* HAL 句柄（指向 Board 提供的实例） */
    uint8_t rx[UART_RX_BUF_SIZE];   /* 接收环形缓冲 */
    uint8_t rx_byte;                /* 单字节 IT 接收的目标 */
    volatile uint16_t head;         /* 写索引（ISR 更新） */
    volatile uint16_t tail;         /* 读索引（任务更新） */
    unsigned char used;             /* 1=已占用 0=空闲 */
} mcu_uart_t;

/* 实例池：每路 UART 一个实例，按 HAL 句柄关联 */
static mcu_uart_t instances[UART_MCU_MAX];

/* =====================================================================
 * ③ 实例管理
 * ===================================================================== */

/* 按 HAL 句柄找已占用的实例 */
static mcu_uart_t *find_instance(const void *hw)
{
    int i;

    for (i = 0; i < UART_MCU_MAX; i++)
        if (instances[i].used && instances[i].handle == hw)
            return &instances[i];
    return NULL;
}

/* 分配一个空闲实例 */
static mcu_uart_t *alloc_instance(void)
{
    int i;

    for (i = 0; i < UART_MCU_MAX; i++) {
        if (!instances[i].used) {
            memset(&instances[i], 0, sizeof(instances[i]));
            instances[i].used = 1U;
            return &instances[i];
        }
    }
    return NULL;
}

/* 由 uart_t.backend（存的是实例指针）反查实例 */
static mcu_uart_t *instance_from_uart(const uart_t *uart)
{
    int i;

    for (i = 0; i < UART_MCU_MAX; i++)
        if ((uintptr_t)&instances[i] == uart->backend &&
            instances[i].used)
            return &instances[i];
    return NULL;
}

/* =====================================================================
 * ④ HAL 状态 → 统一错误码
 * ===================================================================== */
static int hal_result(HAL_StatusTypeDef status)
{
    if (status == HAL_OK)
        return COMMON_OK;
    if (status == HAL_BUSY)
        return COMMON_EBUSY;
    if (status == HAL_TIMEOUT)
        return COMMON_ETIMEOUT;
    return COMMON_EIO;
}

/* =====================================================================
 * ⑤ 接收环形缓冲
 *
 * 单生产者（ISR）/ 单消费者（任务），head/tail 都 volatile。
 * 满时丢新字节（不做丢旧，避免覆盖未读数据）；丢字节策略是
 * 明确选择，统计留给后续扩展。
 * ===================================================================== */
static void rx_push(mcu_uart_t *instance, uint8_t byte)
{
    uint16_t next = (uint16_t)((instance->head + 1U) % UART_RX_BUF_SIZE);

    if (next != instance->tail) {
        instance->rx[instance->head] = byte;
        instance->head = next;
    }
    /* 缓冲满时丢弃新字节；统计策略留给后续扩展。 */
}

/* =====================================================================
 * ⑥ 后端实现（作为 uart_default_ops 的四个方法）
 * ===================================================================== */
static int mcu_open(uart_t *uart, const void *hw, uart_baud_t baud)
{
    UART_MCU_HANDLE_T *handle = (UART_MCU_HANDLE_T *)hw;
    mcu_uart_t *instance;
    int result;

    /* 参数校验 */
    if (handle == NULL || baud == 0U)
        return COMMON_EINVAL;
    /* 同一句柄不允许被重复打开 */
    if (find_instance(hw) != NULL)
        return COMMON_EBUSY;

    /* 分配实例 */
    instance = alloc_instance();
    if (instance == NULL)
        return COMMON_ENOSPC;
    instance->handle = handle;

    /* 8N1、无流控、16 倍过采样；BaudRate 直接用公共波特率常量 */
    handle->Init.BaudRate = baud;
    handle->Init.WordLength = UART_WORDLENGTH_8B;
    handle->Init.Parity = UART_PARITY_NONE;
    handle->Init.StopBits = UART_STOPBITS_1;
    handle->Init.Mode = UART_MODE_TX_RX;
    handle->Init.HwFlowCtl = UART_HWCONTROL_NONE;
    handle->Init.OverSampling = UART_OVERSAMPLING_16;

    result = hal_result(HAL_UART_Init(handle));
    if (result < 0)
        goto fail;
    /* 启动单字节 IT 接收；失败时反初始化并释放实例 */
    result = hal_result(HAL_UART_Receive_IT(handle, &instance->rx_byte, 1U));
    if (result < 0) {
        (void)HAL_UART_DeInit(handle);
        goto fail;
    }

    uart->backend = (uintptr_t)instance;
    return COMMON_OK;

fail:
    /* 统一回滚：归还实例（下次打开可复用） */
    memset(instance, 0, sizeof(*instance));
    return result;
}

static int mcu_write(uart_t *uart, const void *buf, uart_len_t len)
{
    mcu_uart_t *instance = instance_from_uart(uart);
    int result;

    /* HAL 发送长度是 uint16_t，先做上限检查 */
    if (instance == NULL || len > UINT16_MAX)
        return COMMON_EINVAL;
    /* 阻塞发送，带超时 */
    result = hal_result(HAL_UART_Transmit(instance->handle,
                                         (uint8_t *)buf,
                                         (uint16_t)len,
                                         UART_MCU_TX_TIMEOUT_MS));
    return result < 0 ? result : (int)len;
}

static int mcu_read(uart_t *uart, void *buf, uart_len_t len)
{
    mcu_uart_t *instance = instance_from_uart(uart);
    uint8_t *bytes = (uint8_t *)buf;
    uart_len_t count = 0U;

    if (instance == NULL)
        return COMMON_ESTATE;
    /* 非阻塞：尽量取到 len 字节或缓冲取空为止 */
    while (count < len && instance->tail != instance->head) {
        bytes[count++] = instance->rx[instance->tail];
        instance->tail =
            (uint16_t)((instance->tail + 1U) % UART_RX_BUF_SIZE);
    }
    return (int)count;
}

static int mcu_close(uart_t *uart)
{
    mcu_uart_t *instance = instance_from_uart(uart);
    int result;

    if (instance == NULL)
        return COMMON_OK;
    /* 先停 IT 接收，再反初始化；最后清空实例与 backend */
    (void)HAL_UART_AbortReceive(instance->handle);
    result = hal_result(HAL_UART_DeInit(instance->handle));
    memset(instance, 0, sizeof(*instance));
    uart->backend = 0U;
    return result;
}

/* =====================================================================
 * ⑦ 中断接收回调入口
 *
 * 由 HAL_UART_RxCpltCallback 转发：把 rx_byte 压入环形缓冲，
 * 然后立刻重启下一字节 IT 接收（单字节续传模式）。
 * ISR 中只做极简工作，不做任何阻塞/分配/复杂解析。
 * ===================================================================== */
void uart_mcu_rx_complete(const void *hw)
{
    mcu_uart_t *instance = find_instance(hw);

    if (instance == NULL)
        return;
    rx_push(instance, instance->rx_byte);
    (void)HAL_UART_Receive_IT(instance->handle,
                             &instance->rx_byte, 1U);
}

/* =====================================================================
 * ⑧ 默认后端表
 * ===================================================================== */
const uart_ops_t uart_default_ops = {
    mcu_open,
    mcu_write,
    mcu_read,
    mcu_close
};
```

````markdown
# UART 驱动（简化版）

当前版本只解决基本问题：同一套接口完成打开、发送、接收和关闭，并可选地按名称管理多个 UART。

## 1. 结构

```text
应用
  ↓ uart_open / uart_write / uart_read / uart_close
UART 公共层
  ↓ uart_ops_t
Linux termios  或  STM32 HAL
```

| 文件 | 作用 |
| --- | --- |
| `uart.h` | 公共类型和基本 API |
| `uart.c` | 参数检查和后端调用 |
| `uart_linux.c` | Linux 串口实现 |
| `uart_mcu.c` | STM32 HAL 实现 |
| `uart_dev.c` | 可选的具名设备适配 |

Linux 与 STM32 后端二选一参与链接。

## 2. 基本接口

```c
uart_t uart = UART_INITIALIZER;
uint8_t data[64];
int count;

if (uart_open(&uart, hardware, UART_BAUD_115200) != COMMON_OK)
    return;

(void)uart_write(&uart, "hello", 5U);
count = uart_read(&uart, data, sizeof(data));

(void)uart_close(&uart);
```

返回规则：

- `open/close`：成功返回 `COMMON_OK`；
- `read/write`：成功返回实际字节数，失败返回负错误码；
- Linux 无数据时 `read` 返回 0；
- Linux `write` 可能只发送一部分，重要协议应由上层循环发送。

## 3. STM32 接入

GPIO、时钟、NVIC 和 `UART_HandleTypeDef` 由 Board/CubeMX 配置，驱动不硬编码具体 USART 和引脚。

```c
static uart_t debug_uart = UART_INITIALIZER;

void app_init(void)
{
    (void)uart_open(&debug_uart, &huart1, UART_BAUD_115200);
}

void HAL_UART_RxCpltCallback(UART_HandleTypeDef *huart)
{
    uart_mcu_rx_complete(huart);
}
```

HAL 的接收目标是驱动实例中的 `rx_byte` RAM 变量。回调将字节放入环形缓冲，再启动下一字节接收。

## 4. 可选的具名管理

```c
dev_entry_t *console = NULL;

if (uart_dev_register("uart-console", &huart1, &console) == COMMON_OK) {
    (void)uart_dev_open(console, UART_BAUD_115200);
    (void)uart_dev_write(console, "boot\r\n", 6U);
}

/* 停止阶段 */
(void)uart_dev_unregister(console);
```

只有确实需要按名称查找多个设备时，才使用 `uart_dev_*`；单串口程序直接使用 `uart_*` 即可。

## 5. 代码分割与注释约定

每个文件用 `/* ==== ① xxx ==== */` 分节，公共接口带`@brief/@param/@return/@note`，内部函数只注释关键“为什么”：

- `uart.h`：① 类型 → ② 波特率 → ③ 超时 → ④ 对象 → ⑤ 操作表 → ⑥ 结构体 → ⑦ 基础接口 → ⑧ 具名适配 → ⑨ MCU 回调 → ⑩ 扩展预留；
- `uart.c`：① 初始化 → ② 打开(指定表) → ③ 打开(默认) → ④ 发送 → ⑤ 接收 → ⑥ 关闭；
- `uart_dev.c`：① 槽位池 → ② 反查 → ③ 注册 → ④ 开读写关 → ⑤ 注销；
- `uart_linux.c`：① 波特率表 → ② 工具 → ③ 后端实现 → ④ 默认表；
- `uart_mcu.c`：① 适配区 → ② 结构 → ③ 实例管理 → ④ 错误映射 → ⑤ 环形缓冲 → ⑥ 后端实现 → ⑦ 回调 → ⑧ 默认表。
````
