```c_cpp
#ifndef UART_H
#define UART_H

/**
 * @file    uart.h
 * @brief   通用串口(UART)驱动 —— 接口层 + MCU 硬件适配配置区（仅标准 C）
 *
 * ═══════════ 一、分层设计 ═══════════
 *   应用层（你的 main.c）  只调本文件接口，不关心底层硬件
 *   通用核心（uart.c）     包装函数，把调用转发到后端操作表 uart_ops_t
 *   设备层（uart_dev.c）   把设备注册表（通用接口管理）接到本驱动
 *   后端（二选一参与编译） 实现 uart_default_ops：
 *     - uart_linux.c  嵌入式 Linux（POSIX termios）
 *     - uart_mcu.c    STM32 MCU（HAL 库，换芯片只改配置宏）
 *   换平台 = 换一个后端 .c，应用代码一行不改。
 *
 * ═══════════ 二、工程依赖「通用接口管理」片段 ═══════════
 *   通用类型定义（uart_baud_t / uart_len_t / 波特率目录 / UART_DEFAULT_BAUD）
 *   在通用接口管理的 common.h；设备注册表 dev_* 在该片段的 device.h / device.c。
 *   两个片段一起导入工程（或拷入同目录），本驱动只是"使用方"。
 *   本文件只保留与串口强相关的内容：
 *   ① MCU「硬件适配配置区」（② 段）与 ② uart 句柄 / 操作表 / API。
 *
 * ═══════════ 三、命名约定 ═══════════
 *   uart_xxx             公开跨平台 API / 类型（应用层可见）
 *   uart_default_ops     后端必须导出的默认操作表
 *   uart_mcu_xxx         MCU 后端公开符号（如中断入口 uart_mcu_rx_byte）
 *   uart_dev_xxx         设备层适配公开符号（uart_dev.c）
 *   mcu_xxx / linux_xxx  后端文件内私有静态函数
 *
 * ═══════════ 四、快速上手 ═══════════
 *   uart_t u;
 *   uart_open (&u, "/dev/ttyUSB0", UART_DEFAULT_BAUD); // Linux：设备节点
 *   uart_open (&u, &huart1,        UART_DEFAULT_BAUD); // STM32：HAL 句柄
 *   uart_write(&u, "AT+CGMI\r\n", 9);
 *   int n = uart_read (&u, buf, sizeof(buf));
 *   uart_close(&u);
 *
 *   接口取舍：只保留 open/read/write/close 4 个；
 *   set_config / printf / avail / flush 已在 uart_ops_t 注释中预留。
 */

#include <stddef.h>
#include <stdint.h>
#include "common.h"   /* 通用接口管理：uart_baud_t / uart_len_t / 波特率目录等 */
#include "device.h"   /* 通用接口管理：dev_entry_t / dev_* 注册表 */

/* ------------------------------------------------------------------ */
/* 句柄类型：应用层只持有指针，内部实例由各后端自行定义（不透明封装）     */
/* ------------------------------------------------------------------ */
typedef struct uart uart_t;

/* ------------------------------------------------------------------ */
/* 后端操作表：每个后端 .c 实现以下函数指针并导出 uart_default_ops        */
/* ------------------------------------------------------------------ */
typedef struct {
    /** 打开串口：0=成功，-1=失败 */
    int (*open)(uart_t *u, const void *hw, uart_baud_t baud);

    /** 发送 len 字节：返回实际发送字节数，失败返回 -1 */
    int (*write)(uart_t *u, const void *buf, uart_len_t len);

    /** 非阻塞接收：返回实际接收字节数（无数据返回 0），失败返回 -1 */
    int (*read)(uart_t *u, void *buf, uart_len_t len);

    /** 关闭串口并释放资源（无返回值） */
    void (*close)(uart_t *u);

    /* ============ 预留扩展（按需实现后在此追加函数指针） ============
     * int (*set_config)(uart_t*, uint8_t data_bits, char parity,
     *                   uint8_t stop_bits, uint8_t blocking);  数据位/校验/停止位/阻塞
     * int (*printf)(uart_t*, const char *fmt, ...);            格式化发送
     * int (*avail)(uart_t*);                                   接收缓冲可读字节数
     * int (*flush)(uart_t*);                                   清空接收缓冲
     * ============================================================== */
} uart_ops_t;

/* 应用层句柄的实际结构 */
struct uart {
    const uart_ops_t *ops; /* 后端操作表（open 时绑定） */
    void *hw;              /* 平台句柄：Linux=文件描述符；STM32=HAL 句柄指针 */
};

/* 后端导出的默认操作表（uart_linux.c 或 uart_mcu.c 中定义其一） */
extern const uart_ops_t uart_default_ops;

/* ------------------------------------------------------------------ */
/* 应用层统一接口（仅此 4 个，所有平台用法一致）                          */
/* ------------------------------------------------------------------ */

/**
 * @brief  打开串口（默认 8N1、非阻塞读）
 * @param  u    应用层句柄（由调用方提供存储空间）
 * @param  hw   平台句柄：
 *              - Linux：设备节点字符串，如 "/dev/ttyUSB0"；
 *              - STM32：HAL 句柄指针，如 &huart1
 * @param  baud 波特率（可填 UART_DEFAULT_BAUD 宏），如 9600 / 115200 / 230400
 * @return 0=成功；-1=失败
 */
int uart_open(uart_t *u, const void *hw, uart_baud_t baud);

/**
 * @brief  指定后端打开串口（进阶接口，设备管理等按设备选后端时使用）
 * @param  u    句柄
 * @param  ops  后端操作表（如 &uart_default_ops，或设备表里登记的定制后端）
 * @param  hw   平台句柄（同 uart_open）
 * @param  baud 波特率（可填 UART_DEFAULT_BAUD 宏）
 * @return 0=成功；-1=失败
 * @note   uart_open 等价于 uart_open_ops(u, &uart_default_ops, hw, baud)
 */
int uart_open_ops(uart_t *u, const uart_ops_t *ops, const void *hw, uart_baud_t baud);

/**
 * @brief  发送数据（阻塞，全部发出或超时）
 * @param  u   句柄
 * @param  buf 数据缓冲区
 * @param  len 字节数
 * @return 实际发送字节数；失败返回 -1
 */
int uart_write(uart_t *u, const void *buf, uart_len_t len);

/**
 * @brief  接收数据（非阻塞：读空即返回）
 * @param  u   句柄
 * @param  buf 接收缓冲区
 * @param  len 缓冲区容量（字节）
 * @return 实际接收字节数（无数据时 0）；失败返回 -1
 */
int uart_read(uart_t *u, void *buf, uart_len_t len);

/**
 * @brief  关闭串口，释放资源
 * @param  u 句柄（关闭后仍可再次 open）
 */
void uart_close(uart_t *u);

/* ------------------------------------------------------------------ */
/* 设备层适配（uart_dev.c）：把设备注册表（通用接口管理）接到本驱动        */
/* 用法：                                                              */
/*   dev_entry_t *gps = uart_dev_register("uart-gps", &huart1);       */
/*   uart_dev_open(gps, UART_DEFAULT_BAUD);                           */
/*   int n = uart_dev_read(gps, line, sizeof(line));                  */
 /*   uart_dev_unregister("uart-gps");   // 先关、再注销、回收槽          */
/* ------------------------------------------------------------------ */

/**
 * @brief  注册一台具名串口设备（登记进通用注册表 + 分配本驱动实例槽）
 * @param  name 设备名（需与 dev_register 同名规则一致：唯一、非空）
 * @param  hw   平台句柄（同 uart_open 的 hw）
 * @return 设备表项；重名 / 槽满返回 NULL
 * @note   注册不打开；打开的返回：绑定登记时的后端（默认 uart_default_ops）
 */
dev_entry_t *uart_dev_register(const char *name, const void *hw);

/** @brief 打开（等价按设备登记的后端调 uart_open_ops）；0=成功；-1=失败 */
int uart_dev_open(dev_entry_t *d, uart_baud_t baud);

/** @brief 便捷发送（转发到该设备的 uart 句柄）；返回实际发送字节数 */
int uart_dev_write(dev_entry_t *d, const void *buf, uart_len_t len);

/** @brief 便捷接收（非阻塞）；返回实际接收字节数 */
int uart_dev_read(dev_entry_t *d, void *buf, uart_len_t len);

/** @brief 关闭设备（转发到 uart_close） */
void uart_dev_close(dev_entry_t *d);

/**
 * @brief  注销设备：先关（若已打开）→ 通用注册表回收槽位 → 释放实例槽
 * @param  name 设备名；未注册的名字调用无副作用
 */
void uart_dev_unregister(const char *name);

/* ==================== ② MCU 硬件适配配置区 ★换硬件只改这里★ ====================

/**
 * @file    uart.c
 * @brief   串口驱动 · 通用核心（跨平台包装层，任何平台原样编译）
 *
 * 职责：
 *   1. uart_open() 时把编译期选定的后端操作表 uart_default_ops 绑定到句柄；
 *   2. 其余接口做空指针防护后，把调用转发给操作表对应函数指针。
 * 本文件不包含任何平台代码。
 */
#include "uart.h"

#include <string.h> /* memset */

int uart_open_ops(uart_t *u, const uart_ops_t *ops, const void *hw, uart_baud_t baud)
{
    /* 参数校验：句柄 / 后端 / 硬件句柄为空直接失败，避免后续空指针解引用 */
    if (!u || !ops || !ops->open || !hw)
        return -1;

    /* 先清零再绑定：
     * - 保证 ops / hw 初始值确定；
     * - 重复 open（未 close）时不残留上一次的旧状态。 */
    memset(u, 0, sizeof(*u));
    u->ops = ops; /* 绑定调用方指定的后端（设备管理等场景可按设备选后端） */

    return ops->open(u, hw, baud);
}

int uart_open(uart_t *u, const void *hw, uart_baud_t baud)
{
    /* 便捷封装：绑定链接期默认后端（uart_default_ops） */
    return uart_open_ops(u, &uart_default_ops, hw, baud);
}

int uart_write(uart_t *u, const void *buf, uart_len_t len)
{
    /* 未 open（ops 为空）、后端未实现、或 len>0 却无缓冲区 → 保守返回 -1 */
    if (!u || !u->ops || !u->ops->write || (len > 0 && !buf))
        return -1;

    return u->ops->write(u, buf, len);
}

int uart_read(uart_t *u, void *buf, uart_len_t len)
{
    if (!u || !u->ops || !u->ops->read || (len > 0 && !buf))
        return -1;

    return u->ops->read(u, buf, len);
}

void uart_close(uart_t *u)
{
    if (u && u->ops && u->ops->close)
        u->ops->close(u);
}

/**
 * @file    uart_dev.c
 * @brief   串口驱动 × 设备注册表（通用接口管理）的适配层
 *
 * 在通用设备注册表（dev_*）之上提供串口驱动的具名设备用法：
 *   dev_entry_t *gps = uart_dev_register("uart-gps", &huart1);  // 注册+占实例槽
 *   dev_entry_t *d   = dev_find("uart-gps");                    // 按名取设备
 *   uart_dev_open (d, UART_DEFAULT_BAUD);                       // 打开
 *   int n = uart_dev_read(d, line, sizeof(line));
 *   uart_dev_unregister("uart-gps");    // 先关 → 注销 → 槽位回收
 *
 * 依赖：通用接口管理的 common.h / device.h 与 uart.h 同工程编译。
 * 每台已注册 uart 设备占用本文件一个实例槽（静态数组，上限 DEV_MAX）。
 *
 * 对外公开符号：uart_dev_*（uart.h 声明，共 6 个）。
 * 文件内私有静态函数：uart_devc_* 前缀。
 */

#include <string.h>
#include "common.h"
#include "device.h"
#include "uart.h"

/* ---- 实例槽池：每台已注册 uart 设备一个槽（含占用标记） ---- */
typedef struct
{
    uart_t  u;       /* 真实实例（ops 在 open 时绑定） */
    uint8_t used;    /* 1=已分配给某设备（不随 open/close 清零） */
} uart_dev_slot_t;

static uart_dev_slot_t uart_dev_slots[DEV_MAX];

/* ---------------- 文件内私有工具 ---------------- */

/** 找一个空闲实例槽；槽满返回 -1 */
static int uart_devc_alloc(void)
{
    int k;

    for (k = 0; k < DEV_MAX; k++)
        if (!uart_dev_slots[k].used)
            return k;
    return -1;
}

/** 按实例指针释放槽（清零整槽，槽位可复用） */
static void uart_devc_release(const uart_t *u)
{
    int k;

    if (!u)
        return;
    for (k = 0; k < DEV_MAX; k++)
        if (&uart_dev_slots[k].u == u)
        {
            memset(&uart_dev_slots[k], 0, sizeof(uart_dev_slots[k]));
            return;
        }
}

/* ---------------- 设备层 API ---------------- */

dev_entry_t *uart_dev_register(const char *name, const void *hw)
{
    int k = uart_devc_alloc();
    dev_entry_t *d;

    if (k < 0)
        return NULL;            /* 实例槽满 */
    d = dev_register(name, &uart_default_ops, hw, &uart_dev_slots[k].u);
    if (!d)
    {
        uart_devc_release(&uart_dev_slots[k].u);   /* 注册被拒（重名/表满）→ 还槽 */
        return NULL;
    }
    uart_dev_slots[k].used = 1;
    return d;
}

int uart_dev_open(dev_entry_t *d, uart_baud_t baud)
{
    if (!d || !d->used || !d->priv)
        return -1;
    /* 按设备登记的后端打开（默认 uart_default_ops，支持定制后端） */
    return uart_open_ops((uart_t *)d->priv, (const uart_ops_t *)d->ops, d->hw, baud);
}

int uart_dev_write(dev_entry_t *d, const void *buf, uart_len_t len)
{
    if (!d || !d->priv)
        return -1;
    return uart_write((uart_t *)d->priv, buf, len);
}

int uart_dev_read(dev_entry_t *d, void *buf, uart_len_t len)
{
    if (!d || !d->priv)
        return -1;
    return uart_read((uart_t *)d->priv, buf, len);
}

void uart_dev_close(dev_entry_t *d)
{
    if (!d || !d->priv)
        return;
    uart_close((uart_t *)d->priv);
}

void uart_dev_unregister(const char *name)
{
    dev_entry_t *d = dev_find(name);

    if (d)
    {
        uart_dev_close(d);      /* 先关（未打开也无副作用） */
        uart_devc_release((uart_t *)d->priv);
        dev_unregister(name);   /* 再注销注册表槽位 */
    }
}

## Fragment: uart_linux.c
```c_cpp
/**
 * @file    uart_linux.c
 * @brief   串口驱动 · 嵌入式 Linux 后端（POSIX termios）
 *
 * 实现 uart_default_ops；应用层代码见 uart.h，无需改动。
 *
 * 特性：
 *   - 默认 8N1、原始模式（无回显 / 无行缓冲 / 无信号产生）；
 *   - 非阻塞读：VMIN=0 / VTIME=1，最多等 0.1s，无数据立即返回 0；
 *   - 接收缓冲由内核 tty 层提供（flip buffer + 行规程），驱动层无需自建环形缓冲；
 *   - 设备节点：/dev/ttySx（片上 UART）、/dev/ttyUSBx（USB 转串）、
 *               /dev/ttyAMAx、/dev/ttyTHSx 等；
 *   - 依赖：仅标准 C + POSIX（open / termios / read / write）。
 *
 * 编译：gcc uart.c uart_linux.c main.c -o app
 *
 * 对外公开符号：仅 uart_default_ops（uart.h 声明）；其余符号一律 static。
 */
#include "uart.h"

/* 平台特性宏：Linux 需完整 termios 特性；macOS 需 BSD 定义 */
#if defined(__linux__)
#define _GNU_SOURCE
#elif defined(__APPLE__)
#define _DARWIN_C_SOURCE
#endif

#include <errno.h>
#include <fcntl.h>
#include <stdint.h>
#include <termios.h>
#include <unistd.h>

/**
 * @brief  波特率数值 → termios 常量
 * @param  baud 波特率，如 115200
 * @return termios 的 Bxxx 常量；不支持时返回 B0 并置 errno=EINVAL
 * @note   表很小（≤31 项），线性查表即可；Bxxx 是 POSIX 专有常量：
 *         B460800 及以上仅 Linux 定义（macOS/POSIX 无），故高速档条目用
 *         #ifdef __linux__ 包含。换/加波特率：先在 uart.h 目录加 #define，
 *         再在下面表里加一行。
 */
/* 波特率映射表：数值 → termios Bxxx 常量（嵌入式常规静态常量查表，直观可调试） */
static const struct
{
    uart_baud_t num;  /* uart.h 目录中的波特率数值 */

/**
 * @brief  打开串口设备并配置为 8N1 原始模式
 * @param  u    句柄
 * @param  hw   设备节点路径字符串，如 "/dev/ttyUSB0"
 * @param  baud 波特率
 * @return 0=成功；-1=失败（errno 说明原因）
 */
static int linux_open(uart_t *u, const void *hw, uart_baud_t baud)
{
    const char *dev = (const char *)hw;

    if (!dev)
    {
        errno = EINVAL;
        return -1;
    }

    /* 1) 打开设备节点：
     *    O_RDWR    读写；
     *    O_NOCTTY  不把设备当作控制终端（防止被键盘信号干扰）；
     *    O_NDELAY  打开时不等待 DCD 载波（避免挂起）。 */
    int fd = open(dev, O_RDWR | O_NOCTTY | O_NDELAY);
    if (fd < 0)
        return -1;

    /* 2) 恢复阻塞模式：
     *    打开参数里的 O_NDELAY 只是为了不等 DCD；真正的读/写行为
     *    由 termios 的 VMIN/VTIME 决定（见第 6 步），所以这里把它清掉。 */
    if (fcntl(fd, F_SETFL, 0) < 0)
    {
        close(fd);
        return -1;
    }

    /* 3) 读取当前 termios 配置，在其基础上修改 */
    struct termios t;
    if (tcgetattr(fd, &t) != 0)
    {
        close(fd);
        return -1;
    }

    /* 4) 原始模式：一次关掉回显(ECHO)、行缓冲(ICANON)、
     *    信号生成(ISIG)、软流控(IXON/IXOFF) 等所有"终端化"行为 */
    cfmakeraw(&t);

    /* 5) 固定 8N1、关闭硬件流控 */
    t.c_cflag &= ~(CSIZE | PARENB | PARODD | CSTOPB | CRTSCTS);
    t.c_cflag |= CS8;

    /* 波特率：不支持的值直接失败 */
    speed_t spd = linux_baud_to_speed(baud);
    if (spd == B0)
    {
        close(fd);
        return -1;
    }
    if (cfsetispeed(&t, spd) || cfsetospeed(&t, spd))
    {
        close(fd);
        return -1;
    }

    /* 6) 非阻塞读：VMIN=0（不要求凑满字节数）+ VTIME=1（最多等 0.1s），
     *    之后 read() 无数据时返回 0，不会卡死主循环 */
    t.c_cc[VMIN] = 0;
    t.c_cc[VTIME] = 1;

    /* 7) TCSANOW 立即生效，并清空收发缓冲里的旧数据 */
    if (tcsetattr(fd, TCSANOW, &t) != 0)
    {
        close(fd);
        return -1;
    }
    tcflush(fd, TCIOFLUSH);

    /* 8) 文件描述符存入句柄（void* 容纳 int 的保证来自 intptr_t） */
    u->hw = (void *)(intptr_t)fd;
    return 0;
}

/**
 * @brief  发送 len 字节
 * @return 实际写入字节数（通常等于 len）；失败返回 -1
 */
static int linux_write(uart_t *u, const void *buf, uart_len_t len)
{
    return (int)write((int)(intptr_t)u->hw, buf, len);
}

/**
 * @brief  非阻塞接收
 * @return 实际读取字节数（无数据返回 0）；失败返回 -1
 */
static int linux_read(uart_t *u, void *buf, uart_len_t len)
{
    return (int)read((int)(intptr_t)u->hw, buf, len);
}

/**
 * @brief  关闭设备节点并置空句柄
 */
static void linux_close(uart_t *u)
{
    if (u->hw)
    {
        close((int)(intptr_t)u->hw);
        u->hw = NULL;
    }
}

/* 默认操作表：本后端在链接期的唯一公开符号 */
const uart_ops_t uart_default_ops = {
    .open  = linux_open,
    .write = linux_write,
    .read  = linux_read,
    .close = linux_close,
};

/**
 * @file    uart_mcu.c
 * @brief   串口驱动 · MCU 后端（HAL 库，跨芯片可移植版）—— 纯实现
 *
 * ═══════════ 移植原则 ═══════════
 * 所有"换硬件"的开关（芯片 HAL 头 / 句柄类型 / 引脚 / 时钟使能 /
 * 复用功能号 / 实例数 / 环形缓冲大小）都集中在 uart.h 的
 * 「② MCU 硬件适配配置区」。本文件是纯实现，**移植时不需要改动**。
 *
 * ═══════════ 中断接收数据流 ═══════════
 *   外部数据 → USART RX 中断
 *     → HAL_UART_Receive_IT() 每收 1 字节触发 HAL_UART_RxCpltCallback()
 *     → uart_mcu_rx_byte() → mcu_rx_push() 压入环形缓冲 rx[]
 *   应用层 uart_read() → mcu_read() → mcu_rx_pop() 从缓冲取出数据
 *   （其他芯片 / 裸机：在 ISR 里直接调 uart_mcu_rx_byte(&hw, byte) 即可，
 *     同样走上面的环形缓冲，不用改本文件源码）
 *
 * ═══════════ 使用 ═══════════
 *   uart_open(&u, &huart1, UART_DEFAULT_BAUD); // huart1 来自 CubeMX
 *   CubeMX 里开对应 USART 的全局中断(NVIC)；GPIO 由本文件自动初始化。
 *
 * ═══════════ 对外公开符号 ═══════════
 *   uart_default_ops   后端操作表（链接期唯一入口，uart.h 声明）
 *   uart_mcu_rx_byte   跨芯片中断入口（uart.h 声明）
 *   其余符号一律 static，仅本文件可见。
 */
#include "uart.h"

/* HAL 头文件：由 uart.h 的 UART_MCU_HAL_HEADER 宏指定（换芯片只改宏） */
#include UART_MCU_HAL_HEADER

/* ==================== 通用实现区（移植无需改动） ==================== */

/* ---------------- 环形缓冲 + 实例表（中断与主循环共享） ---------------- */
typedef struct {
    UART_MCU_HANDLE_T *huart;      /* 绑定的外设句柄（区分实例用） */
    uint8_t rx[UART_RX_BUF_SIZE];  /* 环形缓冲（大小见 uart.h 2.6） */
    volatile uint16_t head;        /* 写指针：ISR 压入后前进 */
    volatile uint16_t tail;        /* 读指针：主循环取出后前进 */
    uint8_t used;                  /* 槽位是否已分配 */
} uart_inst_t;

/* 实例表：静态数组，避免动态内存分配（嵌入式环境安全） */
static uart_inst_t insts[UART_MCU_MAX];

/**
 * @brief  按外设句柄查找已分配的实例
 * @param  hw 外设句柄
 * @return 实例指针；未找到返回 NULL
 */
static uart_inst_t *mcu_inst_find(const void *hw)
{
    for (int i = 0; i < UART_MCU_MAX; i++)
    {
        if (insts[i].used && insts[i].huart == hw)
            return &insts[i];
    }
    return NULL;
}

/**
 * @brief  分配一个空闲实例槽位
 * @return 实例指针；已满（同时使用超过 UART_MCU_MAX 路串口）返回 NULL
 */
static uart_inst_t *mcu_inst_alloc(void)
{
    for (int i = 0; i < UART_MCU_MAX; i++)
    {
        if (!insts[i].used)
        {
            insts[i].used = 1;
            insts[i].head = insts[i].tail = 0; /* 清空缓冲 */
            return &insts[i];
        }
    }
    return NULL;
}

/**
 * @brief  向环形缓冲压入 1 字节（仅中断上下文调用）
 * @param  i 实例
 * @param  b 字节
 * @note   单生产者模型：生产端只有中断这一个入口；
 *         容量 = SIZE-1；head 追上 tail 判定为满，满则丢弃新字节
 *         （宁可丢新，也不覆盖尚未被读走的数据）。
 */
static void mcu_rx_push(uart_inst_t *i, uint8_t b)
{
    uint16_t nxt = (i->head + 1) & (UART_RX_BUF_SIZE - 1); /* 2 的幂：位与 = 取模 */
    if (nxt != i->tail)
    {
        i->rx[i->head] = b;
        i->head = nxt;
    }
}

/**
 * @brief  从环形缓冲取出 1 字节（主循环调用）
 * @param  i 实例
 * @param  b 输出：取出的字节
 * @return 1=取到；0=缓冲空
 */
static int mcu_rx_pop(uart_inst_t *i, uint8_t *b)
{
    if (i->head == i->tail) /* 空 */
        return 0;

    *b = i->rx[i->tail];
    i->tail = (i->tail + 1) & (UART_RX_BUF_SIZE - 1);
    return 1;
}

/* ---------------- 硬件初始化（全部由 uart.h 配置区宏驱动） ---------------- */

/**
 * @brief  初始化串口硬件：开外设时钟 + 开 GPIO 时钟 + TX/RX 引脚复用
 * @note   幂等：CubeMX 已初始化也无副作用；裸机工程同样自包含。
 *         所有引脚/时钟/复用号来自 uart.h ② 硬件适配配置区。
 */
static void mcu_hw_init(void)
{
    UART_MCU_CLK_ENABLE();      /* 外设时钟：换芯片 / 时钟线只改 uart.h */
    UART_MCU_GPIO_CLK_ENABLE(); /* GPIO 时钟：换端口时同步改 uart.h */

    /* TX/RX 引脚：复用推挽输出、无上下拉、高速 */
    GPIO_InitTypeDef g;
    g.Pin = UART_MCU_TX_PIN;
    g.Mode = GPIO_MODE_AF_PP;
    g.Pull = GPIO_NOPULL;
    g.Speed = GPIO_SPEED_FREQ_HIGH;
#if UART_MCU_AF
    g.Alternate = UART_MCU_AF; /* F4/H7 等：写复用功能号；F1 无该成员，AF=0 整段被裁掉 */
#endif
    HAL_GPIO_Init(UART_MCU_TX_PORT, &g);

    g.Pin = UART_MCU_RX_PIN;
    HAL_GPIO_Init(UART_MCU_RX_PORT, &g);
}

/* ---------------- 后端 4 接口 ---------------- */

/**
 * @brief  打开串口：初始化硬件 + 配置 8N1 + 启动中断接收
 * @param  u    句柄
 * @param  hw   HAL 句柄指针，如 &huart1
 * @param  baud 波特率
 * @return 0=成功；-1=失败（实例满 / HAL_UART_Init 失败）
 */
static int mcu_open(uart_t *u, const void *hw, uart_baud_t baud)
{
    UART_MCU_HANDLE_T *huart = (UART_MCU_HANDLE_T *)hw;

    if (!huart)
        return -1;

    /* 分配实例槽（每路串口一个，上限 UART_MCU_MAX，见 uart.h 2.6） */
    uart_inst_t *i = mcu_inst_alloc();
    if (!i)
        return -1;
    i->huart = huart;

    /* 先开时钟 + 配 GPIO（宏驱动，幂等）；再固定 8N1、无流控、16 倍过采样 */
    mcu_hw_init();
    huart->Init.BaudRate = baud;
    huart->Init.WordLength = UART_WORDLENGTH_8B;
    huart->Init.Parity = UART_PARITY_NONE;
    huart->Init.StopBits = UART_STOPBITS_1;
    huart->Init.Mode = UART_MODE_TX_RX;
    huart->Init.HwFlowCtl = UART_HWCONTROL_NONE;
    huart->Init.OverSampling = UART_OVERSAMPLING_16;
    if (HAL_UART_Init(huart) != HAL_OK)
    {
        i->used = 0; /* 初始化失败：归还实例槽 */
        return -1;
    }

    /* 启动"每收 1 字节进环形缓冲"的中断接收（数据流见文件头注释）；
     * 失败同样归还槽位，保证 open 失败后句柄干净 */
    if (HAL_UART_Receive_IT(huart, (uint8_t *)&huart->Instance->DR, 1) != HAL_OK)
    {
        i->used = 0;
        return -1;
    }

    u->hw = (void *)hw; /* 全部成功后才绑定句柄，不留半初始化状态 */
    return 0;
}

/**
 * @brief  发送 len 字节（阻塞，最多等 1000ms）
 * @return 实际发送字节数；失败返回 -1
 */
static int mcu_write(uart_t *u, const void *buf, uart_len_t len)
{
    return HAL_UART_Transmit((UART_MCU_HANDLE_T *)u->hw, (const uint8_t *)buf, len, 1000) == HAL_OK
               ? (int)len
               : -1;
}

/**
 * @brief  非阻塞读环形缓冲
 * @return 实际读出字节数（无数据返回 0）；实例不存在返回 -1
 */
static int mcu_read(uart_t *u, void *buf, uart_len_t len)
{
    uart_inst_t *i = mcu_inst_find(u->hw);
    if (!i)
        return -1;

    uint8_t *p = (uint8_t *)buf;
    uart_len_t n = 0;
    while (n < len) /* 读满或读空即停（非阻塞） */
    {
        uint8_t b;
        if (!mcu_rx_pop(i, &b))
            break;
        p[n++] = b;
    }
    return n;
}

/**
 * @brief  关闭：停止中断接收，归还实例槽
 */
static void mcu_close(uart_t *u)
{
    uart_inst_t *i = mcu_inst_find(u->hw);
    if (i)
    {
        HAL_UART_AbortReceive((UART_MCU_HANDLE_T *)u->hw); /* 停止中断接收 */
        i->used = 0;                                       /* 归还实例槽 */
    }
    u->hw = NULL;
}

/* ---------------- 中断接收入口（与芯片无关） ---------------- */

/**
 * @brief  跨芯片通用中断入口：收到 1 字节 → 压入对应实例的环形缓冲
 * @param  hw 外设句柄（须先 open 过）
 * @param  b  收到的字节
 * @note   任何芯片的 ISR 都可直接调用；STM32 的 HAL 回调内部也是调它。
 */
void uart_mcu_rx_byte(const void *hw, uint8_t b)
{
    uart_inst_t *i = mcu_inst_find(hw);
    if (i)
        mcu_rx_push(i, b);
}

/**
 * @brief  STM32 HAL 接收完成回调（CubeMX 全局中断自动走到这里）
 * @note   动作：字节入缓冲 + 重新挂起下一个字节的中断接收。
 *         F1/F4/L4… 各系列 HAL 签名一致，本文件通吃。
 */
void HAL_UART_RxCpltCallback(UART_HandleTypeDef *huart)
{
    uart_mcu_rx_byte((const void *)huart, (uint8_t)(huart->Instance->DR & 0xFF));
    HAL_UART_Receive_IT(huart, (uint8_t *)&huart->Instance->DR, 1); /* 继续收下一个 */
}

/* 默认操作表：本后端在链接期的唯一公开符号 */
const uart_ops_t uart_default_ops = {
    .open  = mcu_open,
    .write = mcu_write,
    .read  = mcu_read,
    .close = mcu_close,
};

| uart_ops_t | 后端操作表：4 个函数指针（open/write/read/close）；预留 set_config / printf / avail / flush |
| uart_default_ops | 后端导出的默认操作表，uart.c 通过它绑定实现 |
| uart_mcu_rx_byte() | MCU 中断接收入口（其他芯片 / 裸机 ISR 直接调用） |

## 3. 输出（返回值）约定

| 返回值 | 含义 | 适用 |
|---|---|---|
| 0 | 操作成功 | uart_open |
| \>0 | 实际收发字节数 | uart_write / uart_read |
| 0 | 暂无数据（非阻塞，**不是错误**） | uart_read |
| -1 | 失败（Linux 置 errno；MCU 由 HAL 状态决定） | 全部 |

## 5. 移植指南（分场景，每段说清"改哪里"）

> 所有适配开关集中在 uart.h 的「② MCU 硬件适配配置区」；下面的代码块就是"改哪里"的示例。

### 5.1 换平台：Linux ↔ STM32

接口不变、应用代码**一行不改**；只换参与编译的后端（.c）和 `uart_open` 第二个参数（设备节点 ↔ HAL 句柄）。

## 6. 接收机制（平台差异）

接收路径按平台区分：

- **Linux：无需自建缓冲**。内核 tty 层自带缓冲（flip buffer + 行规程），硬件字节先落在内核里，`read()` 只是取数，不会像 MCU 寄存器那样被覆盖；`uart_read` 非阻塞（VMIN=0 / VTIME=1，最多等 0.1s），无数据返回 0；
- **STM32 / MCU：必须软件缓冲**。硬件只有约 1 字节的 DR 寄存器，ISR 不立即取走就会被下一个字节覆盖，所以用中断 + 256 字节环形缓冲（`UART_RX_BUF_SIZE`）：
  - 中断入口：CubeMX 全局中断自动走 `HAL_UART_RxCpltCallback`（已实现）；其他芯片 / 裸机在 ISR 里调 `uart_mcu_rx_byte(&hw, byte)` 即可，不用改源码；
  - 最多 `UART_MCU_MAX`（=4）个实例，按 hw 句柄区分（&huart1、&huart2… 各自独立）；
  - 环形缓冲满时丢新字节（rx_push 有尾部保护）。

> 补充：Linux 若长期高速灌数而应用来不及读，内核缓冲耗尽同样会丢字节；此时可在应用层加"读线程 + 环形缓冲"作保险，属可选设计，非驱动必需。

## 8. 注意事项

- 本 runner 环境没有厂商 HAL 头文件：点「检查」时 uart_mcu.c 会报 `stm32f1xx_hal.h not found`，属预期；拷进 STM32 工程即可编译；换芯片改 uart.h「② 硬件适配配置区」的 `UART_MCU_HAL_HEADER`（如 `stm32f4xx_hal.h`）；
- `uart_linux.c` 的 B460800 及以上波特率常量为 Linux 专有，已用 `#ifdef __linux__` 保护，macOS 编译不受影响；
- 本驱动依赖「通用接口管理」片段（common.h / device.h / device.c）：工程内**两个片段都要导入**；点「检查」时 `common.h / device.h not found` 属预期（同 stm32 HAL 头），拷进工程即可；
- Linux 设备节点：`/dev/ttySx`（片上 UART）、`/dev/ttyUSBx`（USB 转串）、`/dev/ttyAMAx`、`/dev/ttyTHSx`；
- 设备节点 / `ptsname()` 等返回**静态缓冲**的字符串，传给 `uart_open` / `uart_dev_register` 前须自行拷贝（第二次调用会覆盖第一次的结果）；
- 默认 8N1；如需数据位/校验/停止位，实现预留的 `set_config` 即可。

## 10. 设备管理（通用注册表 + uart 适配层）

设备注册表 `dev_*` 在「通用接口管理」片段（纯登记、不做 I/O）；本驱动用 uart_dev.c 把它接起来，
补上串口专属的打开 / 收发。业务代码只认名字：**注册 → find → 开发 → 注销**。

### 10.1 接口输入 / 输出

| 接口 | 输入（参数） | 输出（返回值） | 说明 |
|---|---|---|---|
| uart_dev_register | name=设备名, hw=平台句柄 | **dev_entry_t\***；失败 **NULL**（重名/槽满） | 登记进注册表 + 分配实例槽（不打开） |
| dev_find | name=设备名 | **dev_entry_t\***；未注册 **NULL** | 按名取设备（通用注册表） |
| uart_dev_open | d=表项, baud=波特率 | 同 uart_open：**0**/**-1** | 按设备登记的后端打开（内部走 uart_open_ops） |
| uart_dev_write / uart_dev_read | 同 uart_write / uart_read | 同 uart_* | 便捷收发（转发到该设备实例） |
| uart_dev_close / uart_dev_unregister | d / name | 无 | 关闭；注销 = 先关 → dev_unregister → 实例槽回收 |
| dev_count / dev_name | - / seq | 数量 / 名字或 NULL | 枚举设备列表（通用注册表，自动跳空位） |
| uart_open_ops | u, ops, hw, baud | 同 uart_open | 指定后端打开（注册表保存每台设备自己的 ops） |

> 配置：通用接口管理 device.h「① 配置区」—— `DEV_MAX`=8 台 / `DEV_NAME_MAX`=16 字节名；实例槽池同样限 `DEV_MAX`，全部静态数组、无动态内存。

### 10.2 使用示例（注册 → find → 开发）

```c
#include "uart.h"      /* uart.h 内含 common.h / device.h（通用接口管理） */

int main(void)
{
    /* 1) 上电初始化：两路串口登记成具名设备（各自分配实例槽） */
    uart_dev_register("uart-gps", &huart1);        /* STM32：HAL 句柄 */
    uart_dev_register("uart-dbg", "/dev/ttyUSB0"); /* Linux：设备节点 */

    /* 2) 业务代码：只认名字，不关心后端 / 硬件 */
    dev_entry_t *gps = dev_find("uart-gps");
    if (gps && uart_dev_open(gps, UART_DEFAULT_BAUD) == 0)
    {
        uart_dev_write(gps, "AT+CGMI\r\n", 9);
        char line[64];
        int n = uart_dev_read(gps, line, sizeof(line));
    }

    /* 3) 用完注销：先关 → 注销 → 实例槽回收 */
    uart_dev_unregister("uart-dbg");
    return 0;
}
```

### 10.3 说明

- 注册**不打开**（open 由 uart_dev_open 显式完成），便于"先登记、后按需使用"；
- 每台已注册 uart 设备占一个实例槽（uart_dev.c 静态池），槽总数 = `DEV_MAX` = 8，可裁剪；
- 同名重复注册被拒绝；注销后同名可重新注册；`dev_find` 未命中返回 NULL；
- MCU 侧**同时打开**的实例数受 `UART_MCU_MAX`（默认 4）上限约束；
- 每台设备登记时绑定自己的后端（默认 `uart_default_ops`）；定制后端：`dev_register(name, 定制ops, hw, &自己的uart_t)` + `uart_open_ops` 直接操作；
- 通用注册表本身不做 I/O，未来 IIC / SPI 驱动复用同一注册表，各自添加自己的适配层（如 uart_dev.c 之于串口）。
```
