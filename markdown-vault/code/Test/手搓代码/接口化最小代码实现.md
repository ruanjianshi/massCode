```c_cpp
#ifndef LED_H
#define LED_H

#include <stdint.h>
#include <stdio.h>

#define LOGENABLE 1

// 安全保护，日志调试
#if LOGENABLE
#define LOG_ERROR(fmt, ...) printf("[ERROR]" fmt "\r\n", ##__VA_ARGS__)
#define LOG_INFO(fmt, ...) printf("[INFO]" fmt "\r\n", ##__VA_ARGS__)
#else
#define LOG_ERROR(fmt, ...) ((void)0)
#define LOG_INFO(fmt, ...) ((void)0)
#endif

typedef struct led led_t; // 封装对象

typedef struct led_ops {
  int (*open)(led_t *led_handler);
  int (*read)(led_t *led_handler, int *status);
  int (*write)(led_t *led_handler, int target);
  int (*close)(led_t *led_handler);
} led_ops_t;

typedef struct led_config {
  const void *port;
  uint32_t pin;
  uint8_t active;
} led_config_t;

led_t *led_create(led_config_t *config, led_ops_t *ops);

// 应用层
int dev_open(led_t *led_handler);
int dev_read(led_t *led_handler, int *status);
int dev_write(led_t *led_handler, int target);
int dev_close(led_t *led_handler);

#endif
```

## Fragment: led.c
```c_cpp
#include "led.h"

//配置和操作、数据绑定
struct led{
	led_config_t *config;
	led_ops_t *ops;
	uint8_t status;
};


#define STATICPOOLSIZE 1
static led_t pool_array[STATICPOOLSIZE];  //指针数组未分配地址，属于垃圾地址
static int8_t pool_count = 0;

//驱动层
static int led_open(led_t *led_handler);
static int led_read(led_t *led_handler, int* status);
static int led_write(led_t *led_handler, int target);
static int led_close(led_t *led_handler);

//管理层
static led_ops_t default_ops = {led_open,led_read,led_write,led_close};



led_t *led_create(led_config_t* config, led_ops_t* ops){
	if(config == NULL) return NULL;

	if(pool_count < STATICPOOLSIZE){
		pool_array[pool_count].config = config;
		pool_array[pool_count].ops = (ops != NULL) ? ops : &default_ops; 
		pool_array[pool_count].status = 0;
		
		LOG_INFO("成功创建，静态池已有个数 %d", (pool_count + 1));
		return &pool_array[pool_count++]; 
	}
	LOG_ERROR("创建失败，静态池已满，大小 %d", STATICPOOLSIZE);
	return NULL;
}


static int led_open(led_t *led_handler){
	if(led_handler == NULL) return -1;
	
	LOG_INFO("对创建的设备进行初始化");
	led_handler->status = 0;
	return 0;
}

static int led_read(led_t *led_handler, int* status){
	if(led_handler == NULL || status == NULL) return -1;
	
	*status = led_handler->status;
	LOG_INFO("读取led的电平状态 %d", *status);
	return 0;
}

static int led_write(led_t *led_handler,int target){
	if(led_handler == NULL) return -1;
	
	led_handler->status = target;
	LOG_INFO("写入led电平 %d", target);
	return 0;
}

static int led_close(led_t *led_handler){
	if(led_handler == NULL) return -1;
	
	for(int i = 0; i < STATICPOOLSIZE; i++){
		// ❌ error: C 不允许 struct 用 == 比较
		if(&pool_array[i] == led_handler){
			for(int j = i; (j + 1) < pool_count; j++){
				pool_array[j] = pool_array[j + 1];
			}
			pool_array[--pool_count] = (led_t){0};
			LOG_INFO("关闭并释放槽 %d,静态池剩余个数 %d", i, pool_count);
			return 0;
		}
	}
	LOG_ERROR("关闭失败：未找到该设备");
	return -1;
}

//应用层
int dev_open(led_t *led_handler){
	if(led_handler == NULL || led_handler->ops == NULL) return -1;
	return led_handler->ops->open(led_handler);
}
int dev_read(led_t *led_handler, int* status){
	if(led_handler == NULL || led_handler->ops == NULL ||status == NULL) return -1;
	return led_handler->ops->read(led_handler,status);
}
int dev_write(led_t *led_handler, int target){
	if(led_handler == NULL || led_handler->ops == NULL) return -1;
	return led_handler->ops->write(led_handler, target);
}
int dev_close(led_t *led_handler){
	if(led_handler == NULL || led_handler->ops == NULL) return -1;
	return led_handler->ops->close(led_handler);
}
```

## Fragment: main.c
```c_cpp
#include "led.h"
#include <math.h>

static led_config_t red_config = {"GPIOA", 1, 1};

int main() {

  // static 局部变量必须编译期定值，不能放运行期函数调用
  led_t *red = led_create(&red_config, NULL);
  led_t *green = led_create(&red_config, NULL);
  // 这是一种修改
  int status;
  dev_open(red);
  dev_read(red, &status);
  dev_write(red, 1);
  dev_read(red, &status);
  dev_close(red);
  green = led_create(&red_config, NULL);
  return 0;
}
```
